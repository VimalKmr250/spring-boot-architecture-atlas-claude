An unmodifiable view still changes underneath you; an immutable copy never does

Collections.unmodifiableList wraps a list and blocks writes through the wrapper only: the owner of the original can still change it, and the view shows every change. List.of and List.copyOf build lists that nobody can change. They also reject null, which surprises people the first time contains(null) throws. Java 21 gave the ordered collections one shared interface for their ends.

Unmodifiable.java · views, copies, their classes, the sequenced methods
import java.util.*;
import java.util.stream.*;

public class Unmodifiable {
    static String name(Object o) { return o.getClass().getName().replace("java.util.", ""); }

    public static void main(String[] args) {
        List<String> source = new ArrayList<>(List.of("a", "b"));
        List<String> view = Collections.unmodifiableList(source);
        List<String> copy = List.copyOf(source);
        source.add("c");
        System.out.println("after source.add(c): view " + view + ", copy " + copy);
        try { view.add("x"); } catch (UnsupportedOperationException e) { System.out.println("view.add -> UnsupportedOperationException"); }

        try { List.of("x", null); } catch (NullPointerException e) { System.out.println("List.of(\"x\", null) -> NullPointerException"); }
        try { List.of("x").contains(null); } catch (NullPointerException e) { System.out.println("List.of(\"x\").contains(null) -> NullPointerException"); }
        System.out.println("Stream.toList keeps nulls: " + Stream.of("x", null).toList());

        System.out.println("List.of(1, 2)               " + name(List.of(1, 2)));
        System.out.println("List.of(1, 2, 3)            " + name(List.of(1, 2, 3)));
        System.out.println("Stream.toList()             " + name(Stream.of(1).toList()));
        System.out.println("Collectors.toList()         " + name(Stream.of(1).collect(Collectors.toList())) + " (mutable, but not promised)");
        System.out.println("Collections.unmodifiableList " + name(view));
        List<Integer> fixed = List.of(1, 2, 3);
        System.out.println("List.copyOf of an immutable list is the same object: " + (List.copyOf(fixed) == fixed));

        List<String> steps = new ArrayList<>(List.of("placed", "paid", "shipped"));
        System.out.println("getFirst " + steps.getFirst() + ", getLast " + steps.getLast() + ", reversed " + steps.reversed());
        steps.reversed().addFirst("delivered");                       // a view: writes go to steps
        steps.addFirst("created");
        System.out.println("after reversed().addFirst and addFirst: " + steps);

        LinkedHashMap<String, Integer> recent = new LinkedHashMap<>();
        recent.put("b", 2);
        recent.put("c", 3);
        recent.putFirst("a", 1);
        System.out.println("LinkedHashMap " + recent + ", firstEntry " + recent.firstEntry()
                + ", pollLastEntry " + recent.pollLastEntry() + ", reversed keys " + recent.sequencedKeySet().reversed());
        try { List.of().getFirst(); } catch (NoSuchElementException e) { System.out.println("List.of().getFirst() -> NoSuchElementException"); }
    }
}
