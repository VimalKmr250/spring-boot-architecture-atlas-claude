A table can be a reliable job queue: each worker locks the next free rows and skips the ones another worker holds

Outbox relays, payment retries and email sends often keep pending work in a table. Several workers poll it, and each job must go to exactly one worker. With plain SELECT … FOR UPDATE, the second worker waits on the rows the first has locked: one worker at a time. With SKIP LOCKED (Oracle) or the READPAST hint (SQL Server), a worker passes over locked rows and takes the next free ones. Workers then never wait for each other, and never take the same job. The claim and the status change happen in one transaction, so a crashed worker's rows unlock and return to the queue.

import java.sql.*;
import java.util.ArrayList;
import java.util.List;

/** Two workers claim jobs from one table at the same time, without waiting for each other or taking the same job. */
public class WorkQueue {
    public static void main(String[] args) throws Exception {
        oracle();
        sqlServer();
    }

    static void oracle() throws SQLException {
        System.out.println("== Oracle 26ai");
        String url = "jdbc:oracle:thin:@//localhost:1521/FREEPDB1";
        try (Connection a = open(url, "shop", "Atlas_pw1"); Connection b = open(url, "shop", "Atlas_pw1")) {
            setUp(a, "CREATE TABLE jobs (job_id NUMBER PRIMARY KEY, status VARCHAR2(10) NOT NULL, worker VARCHAR2(10))",
                     "INSERT INTO jobs SELECT LEVEL, 'READY', NULL FROM dual CONNECT BY LEVEL <= 8");
            // the obvious query is refused: FETCH FIRST and FOR UPDATE cannot be combined
            try (Statement s = a.createStatement()) {
                s.executeQuery("SELECT job_id FROM jobs WHERE status = 'READY' ORDER BY job_id FETCH FIRST 3 ROWS ONLY FOR UPDATE SKIP LOCKED");
            } catch (SQLException e) {
                System.out.println("  " + e.getMessage().lines().findFirst().orElse(""));
                a.rollback();
            }
            // instead: open a SKIP LOCKED cursor and fetch only 3 rows; rows are locked as they are fetched
            String claim = "SELECT job_id FROM jobs WHERE status = 'READY' ORDER BY job_id FOR UPDATE SKIP LOCKED";
            List<Integer> jobsA = claim(a, claim, "A");               // A holds its jobs, uncommitted
            List<Integer> jobsB = claim(b, claim, "B");               // B skips them instead of waiting
            System.out.println("  worker A claimed " + jobsA + ", worker B claimed " + jobsB);
            a.commit(); b.commit();
            report(a);
            setUp(a, "DROP TABLE jobs PURGE");
        }
    }

    static void sqlServer() throws SQLException {
        System.out.println("== SQL Server 2025");
        String url = "jdbc:sqlserver://localhost:1433;databaseName=shop;encrypt=false";
        try (Connection a = open(url, "sa", "Atlas_pw1!"); Connection b = open(url, "sa", "Atlas_pw1!")) {
            setUp(a, "CREATE TABLE jobs (job_id INT PRIMARY KEY, status VARCHAR(10) NOT NULL, worker VARCHAR(10))",
                     "INSERT INTO jobs SELECT value, 'READY', NULL FROM GENERATE_SERIES(1, 8)");
            // UPDLOCK claims the rows, READPAST skips rows another worker has locked
            String claim = """
                WITH next_jobs AS (
                  SELECT TOP (3) job_id, status, worker FROM jobs WITH (UPDLOCK, READPAST, ROWLOCK)
                  WHERE status = 'READY' ORDER BY job_id)
                UPDATE next_jobs SET status = 'TAKEN', worker = ? OUTPUT inserted.job_id""";
            List<Integer> jobsA = claimInOneStatement(a, claim, "A");
            List<Integer> jobsB = claimInOneStatement(b, claim, "B");
            System.out.println("  worker A claimed " + jobsA + ", worker B claimed " + jobsB);
            a.commit(); b.commit();
            report(a);
            setUp(a, "DROP TABLE jobs");
        }
    }

    // Oracle: lock up to 3 rows through the cursor, then mark them taken in the same transaction
    static List<Integer> claim(Connection c, String sql, String worker) throws SQLException {
        List<Integer> ids = new ArrayList<>();
        try (PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setFetchSize(3);
            try (ResultSet rs = ps.executeQuery()) {
                while (ids.size() < 3 && rs.next()) ids.add(rs.getInt(1));
            }
        }
        try (PreparedStatement ps = c.prepareStatement("UPDATE jobs SET status = 'TAKEN', worker = ? WHERE job_id = ?")) {
            for (int id : ids) { ps.setString(1, worker); ps.setInt(2, id); ps.addBatch(); }
            ps.executeBatch();
        }
        return ids;
    }

    static List<Integer> claimInOneStatement(Connection c, String sql, String worker) throws SQLException {
        List<Integer> ids = new ArrayList<>();
        try (PreparedStatement ps = c.prepareStatement(sql)) {
            ps.setString(1, worker);
            try (ResultSet rs = ps.executeQuery()) { while (rs.next()) ids.add(rs.getInt(1)); }
        }
        ids.sort(null);
        return ids;
    }

    static void report(Connection c) throws SQLException {
        try (Statement s = c.createStatement();
             ResultSet rs = s.executeQuery("SELECT status, COUNT(*) FROM jobs GROUP BY status ORDER BY status")) {
            StringBuilder sb = new StringBuilder("  after both commits:");
            while (rs.next()) sb.append(' ').append(rs.getString(1)).append('=').append(rs.getInt(2));
            System.out.println(sb);
        }
        c.commit();
    }

    static Connection open(String url, String user, String password) throws SQLException {
        Connection c = DriverManager.getConnection(url, user, password);
        c.setAutoCommit(false);
        return c;
    }

    static void setUp(Connection c, String... sql) throws SQLException {
        try (Statement s = c.createStatement()) { for (String q : sql) s.execute(q); }
        c.commit();
    }
}

ORA-02014The natural Oracle query, … FETCH FIRST 3 ROWS ONLY FOR UPDATE SKIP LOCKED, is refused: the row limit turns the query into a view that cannot be locked. Open the cursor without a limit and stop fetching after three rows. JDBC's fetch size of 3 makes the first fetch lock exactly three.
Not a message brokerA table queue gives transactions together with your business data: claiming a payment retry and updating the payment in one commit. It does not give push delivery, fan-out or high throughput. For those, publish from the table through an outbox: see Outbox and CDC.
Say it in the interview"For a table-based queue each worker claims a batch with SKIP LOCKED on Oracle, or READPAST plus UPDLOCK on SQL Server, inside a short transaction, with an index on the status column. Jobs need a lease or retry count, so a job that keeps failing does not loop forever."
