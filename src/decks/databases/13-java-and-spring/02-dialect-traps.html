The same JDBC code can fail, or quietly slow down, depending on which database is behind the DataSource

JDBC and JPA hide most of the differences between databases, but not all of them, and the ones that leak through are found in production. The program below sends identical calls to both engines. A long IN list, which is what findAllById(ids) generates, hits a parameter limit. An empty string becomes NULL on Oracle. An isolation level is refused. And a String parameter arrives at SQL Server as NVARCHAR, which turns an index seek into a scan.

DialectTraps.java · ojdbc17 23.26.3, mssql-jdbc 13.4.0
import java.sql.*;
import java.util.Collections;

/** Four ways the same JDBC code behaves differently on Oracle and SQL Server. */
public class DialectTraps {
    static final String ORACLE = "jdbc:oracle:thin:@//localhost:1521/FREEPDB1";
    static final String MSSQL = "jdbc:sqlserver://localhost:1433;databaseName=shop;encrypt=false";

    public static void main(String[] args) throws SQLException {
        try (Connection ora = DriverManager.getConnection(ORACLE, "shop", "Atlas_pw1");
             Connection mss = DriverManager.getConnection(MSSQL, "sa", "Atlas_pw1!")) {

            // 1  a long IN list, as an ORM builds for findAllById(ids): each engine has a limit
            System.out.println("1 IN list with 65,536 values on Oracle:  " + inList(ora, 65_536));
            System.out.println("  IN list with 2,101 values on SQL Server: " + inList(mss, 2_101));

            // 2  the empty string: Oracle stores NULL, so a NOT NULL column rejects it
            for (Connection c : new Connection[] {ora, mss}) {
                c.setAutoCommit(false);
                try (PreparedStatement ps = c.prepareStatement(
                        "UPDATE customers SET name = ? WHERE customer_id = 10")) {
                    ps.setString(1, "");
                    System.out.println("2 setString(\"\") on " + engine(c) + ": " + ps.executeUpdate() + " row updated");
                } catch (SQLException e) {
                    System.out.println("2 setString(\"\") on " + engine(c) + ": " + firstLine(e));
                }
                c.rollback();
                c.setAutoCommit(true);
            }

            // 3  isolation levels: Oracle accepts only READ COMMITTED and SERIALIZABLE
            try {
                ora.setTransactionIsolation(Connection.TRANSACTION_REPEATABLE_READ);
            } catch (SQLException e) {
                System.out.println("3 REPEATABLE_READ on Oracle: " + firstLine(e));
            }

            // 4  String parameters reach SQL Server as NVARCHAR unless the driver is told otherwise
            try (Statement s = mss.createStatement()) {
                s.execute("CREATE INDEX big_customers_email_ix ON big_customers (email)");
                s.execute("DBCC FREEPROCCACHE WITH NO_INFOMSGS");
            }
            lookups(mss);                                             // default: sendStringParametersAsUnicode=true
            try (Connection ascii = DriverManager.getConnection(MSSQL + ";sendStringParametersAsUnicode=false", "sa", "Atlas_pw1!")) {
                lookups(ascii);
            }
            try (Statement s = mss.createStatement(); ResultSet rs = s.executeQuery("""
                    SELECT SUBSTRING(t.text, 2, CHARINDEX(')', t.text) - 1) AS parameter,
                           qs.execution_count, qs.total_logical_reads / qs.execution_count AS reads_per_lookup
                    FROM   sys.dm_exec_query_stats qs CROSS APPLY sys.dm_exec_sql_text(qs.sql_handle) t
                    WHERE  t.text LIKE '%FROM big_customers WHERE email%' AND t.text NOT LIKE '%dm_exec%'
                    ORDER  BY reads_per_lookup DESC""")) {
                while (rs.next()) System.out.printf("4 %-20s %d lookups, %d page reads each%n",
                        rs.getString(1), rs.getLong(2), rs.getLong(3));
                s.execute("DROP INDEX big_customers_email_ix ON big_customers");
            }
        }
    }

    static String inList(Connection c, int n) {
        String sql = "SELECT COUNT(*) FROM big_customers WHERE customer_id IN (" + String.join(",", Collections.nCopies(n, "?")) + ")";
        try (PreparedStatement ps = c.prepareStatement(sql)) {
            for (int i = 1; i <= n; i++) ps.setInt(i, i);
            try (ResultSet rs = ps.executeQuery()) { rs.next(); return rs.getInt(1) + " rows"; }
        } catch (SQLException e) {
            return firstLine(e);
        }
    }

    static void lookups(Connection c) throws SQLException {
        try (PreparedStatement ps = c.prepareStatement("SELECT customer_id FROM big_customers WHERE email = ?")) {
            for (int i = 1; i <= 100; i++) {
                ps.setString(1, "user" + i + "@example.com");
                try (ResultSet rs = ps.executeQuery()) { rs.next(); }
            }
        }
    }

    static String engine(Connection c) throws SQLException { return c.getMetaData().getDatabaseProductName(); }

    static String firstLine(SQLException e) { return e.getMessage().lines().findFirst().orElse(""); }
}

TrapFix
IN-list limits: 2,100 parameters per statement on SQL Server; 65,535 list items on Oracle 26ai, 1,000 on 19cChunk the ids, join to a temporary table, or pass an array (Oracle MEMBER OF, SQL Server OPENJSON or a table-valued parameter)
'' is NULL on OracleValidate before saving; never use '' as "no value" in portable code
Oracle has no REPEATABLE READREAD COMMITTED with locking or versions, or SERIALIZABLE
Unicode parameters on VARCHAR columnssendStringParametersAsUnicode=false, or NVARCHAR columns throughout
Generated keysName the column: prepareStatement(sql, new String[] {"ID"}) works on both
88 pages instead of 2The same 100 lookups by email cost 88 page reads each with the driver's default, @P0 nvarchar(4000), and 2 each with sendStringParametersAsUnicode=false, @P0 varchar(8000). Nothing fails; the database just works 44 times harder on every login. See Why the optimiser ignores your index for the plan.
Limits move between releasesThe well-known Oracle limit of 1,000 items in an IN list (ORA-01795) is 65,535 on 26ai, as the error message above states. Code that chunks at 1,000 still works everywhere; code written against 26ai's limit fails on 19c. Test against the version you run in production.
