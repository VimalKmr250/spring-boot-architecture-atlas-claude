Spring turns ORA-00001 and SQL Server error 2627 into the same DuplicateKeyException, so retry logic can be written once

Calling a stored procedure from Java differs by engine. An Oracle procedure returns rows through an OUT parameter of type SYS_REFCURSOR, which JDBC reads as a ResultSet. A SQL Server procedure simply returns a result set, so Spring's JdbcClient can treat it like a query. Errors are where Spring helps most. Its translator maps each vendor's error codes onto one hierarchy of DataAccessException subclasses, so a service can catch "duplicate key" or "deadlock loser" without knowing which database it runs on.

ProceduresAndErrors.java · Spring Framework 7.0.9 JdbcClient, ojdbc17, mssql-jdbc
import java.sql.*;
import java.util.List;
import javax.sql.DataSource;

import org.springframework.dao.DataAccessException;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.jdbc.datasource.DriverManagerDataSource;
import org.springframework.jdbc.support.SQLErrorCodeSQLExceptionTranslator;

/** Calling procedures from Java, and the exceptions Spring turns database errors into. */
public class ProceduresAndErrors {
    public static void main(String[] args) throws SQLException {
        DataSource oracle = new DriverManagerDataSource("jdbc:oracle:thin:@//localhost:1521/FREEPDB1", "shop", "Atlas_pw1");
        DataSource mssql = new DriverManagerDataSource("jdbc:sqlserver://localhost:1433;databaseName=shop;encrypt=false", "sa", "Atlas_pw1!");

        // 1  Oracle returns rows from a procedure through an OUT parameter of type SYS_REFCURSOR
        JdbcClient.create(oracle).sql("""
            CREATE OR REPLACE PROCEDURE orders_of(p_customer IN NUMBER, p_orders OUT SYS_REFCURSOR) AS
            BEGIN
              OPEN p_orders FOR SELECT order_id, status FROM orders WHERE customer_id = p_customer ORDER BY order_id;
            END;""").update();
        try (Connection c = oracle.getConnection(); CallableStatement cs = c.prepareCall("{call orders_of(?, ?)}")) {
            cs.setInt(1, 3);
            cs.registerOutParameter(2, Types.REF_CURSOR);
            cs.execute();
            try (ResultSet rs = cs.getObject(2, ResultSet.class)) {
                while (rs.next()) System.out.println("1 Oracle ref cursor: " + rs.getInt(1) + " " + rs.getString(2));
            }
        }

        // 2  SQL Server procedures return result sets directly, so JdbcClient can read them like a query
        JdbcClient sql = JdbcClient.create(mssql);
        sql.sql("CREATE OR ALTER PROCEDURE orders_of @customer INT AS "
              + "SELECT order_id, status FROM orders WHERE customer_id = @customer ORDER BY order_id").update();
        sql.sql("EXEC orders_of ?").param(3).query((rs, n) -> rs.getInt(1) + " " + rs.getString(2)).list()
           .forEach(r -> System.out.println("2 SQL Server result set: " + r));

        // 3  the same three mistakes on both engines, as Spring reports them
        for (DataSource ds : List.of(oracle, mssql)) {
            JdbcClient db = JdbcClient.create(ds);
            String engine = ds == oracle ? "Oracle    " : "SQL Server";
            report(engine, "duplicate email", () -> db.sql(
                "INSERT INTO customers (customer_id, name, email, country, tier, created_at) VALUES (11, 'X', 'asha@example.com', 'IN', 'GOLD', CURRENT_TIMESTAMP)").update());
            report(engine, "unknown customer", () -> db.sql(
                "INSERT INTO orders (customer_id, status, placed_at, total) VALUES (999, 'PENDING', CURRENT_TIMESTAMP, 10)").update());
            report(engine, "zero quantity", () -> db.sql(
                "INSERT INTO order_items (order_id, line_no, product_id, qty, unit_price) VALUES (1001, 3, 103, 0, 29.90)").update());
        }

        // 4  errors that need two sessions to provoke: what Spring maps their vendor codes to
        SQLErrorCodeSQLExceptionTranslator oraT = new SQLErrorCodeSQLExceptionTranslator(oracle);
        SQLErrorCodeSQLExceptionTranslator mssT = new SQLErrorCodeSQLExceptionTranslator(mssql);
        System.out.println("4 ORA-00060 deadlock      -> " + name(oraT.translate("debit", null, new SQLException("deadlock", "61000", 60))));
        System.out.println("  ORA-00054 lock busy     -> " + name(oraT.translate("lock", null, new SQLException("busy", "61000", 54))));
        System.out.println("  1205 deadlock victim    -> " + name(mssT.translate("debit", null, new SQLException("victim", "40001", 1205))));
        System.out.println("  1222 lock time-out      -> " + name(mssT.translate("lock", null, new SQLException("timeout", "HY000", 1222))));

        JdbcClient.create(oracle).sql("DROP PROCEDURE orders_of").update();
        sql.sql("DROP PROCEDURE orders_of").update();
    }

    static void report(String engine, String what, Runnable call) {
        try {
            call.run();
            System.out.println("3 " + engine + " " + what + ": no error");
        } catch (DataAccessException e) {
            System.out.printf("3 %s %-17s -> %s%n", engine, what + ":", name(e));
        }
    }

    static String name(Object e) {
        if (e == null) return "not translated";
        Class<?> c = e.getClass();
        return c.getSimpleName() + " (" + c.getSuperclass().getSimpleName() + ")";
    }
}

How step 4 was producedDeadlocks and lock timeouts need two sessions to provoke; Deadlocks does that for real. Here the translator was handed SQLExceptions carrying those vendor codes, which shows what the same Spring application would throw. Both deadlock codes land on PessimisticLockingFailureException subclasses.
Catching too broadlycatch (DataAccessException e) treats a duplicate key, a deadlock and a broken connection alike. Retrying a constraint violation never succeeds, while not retrying a deadlock loses work. Catch the specific subclasses, and let Spring Retry or Resilience4j retry only the transient ones.
Say it in the interview"I let Spring translate SQL errors and branch on the exception type: DuplicateKeyException for idempotency, PessimisticLockingFailureException for retries, everything else as a failure. For an Oracle ref cursor I register a REF_CURSOR out parameter and read it as a ResultSet."
