Kahn's algorithm is a stream of frontiers; the indegree decrement is the state

Build the graph with collectors: groupingBy for the adjacency, and a collect with a real combiner for the indegrees. Kahn's algorithm then becomes a stream of frontiers. First come the nodes with nothing left to wait for, then the nodes they release, and so on. Exactly one lambda carries state: the filter that decrements an indegree and admits the node when it hits zero. 1462 needs no traversal at all, because its transitive closure is a pure fold over BitSet rows.

LC 207 · Course Schedule · stateful
class Solution {
    public boolean canFinish(int n, int[][] prerequisites) {
        // e = [course, prereq]: the edge runs e[1] -> e[0]
        Map<Integer, List<Integer>> next = adj(Arrays
                .stream(prerequisites)
                .map(e -> new int[]{ e[1], e[0] }));
        int[] waits = Arrays.stream(prerequisites).collect(
                () -> new int[n], (w, e) -> w[e[0]]++,
                (a, b) -> Arrays.setAll(a, i -> a[i] + b[i]));
        List<Integer> free = IntStream.range(0, n)
                .filter(c -> waits[c] == 0).boxed().toList();
        // It works exactly when every course gets released.
        return Stream.iterate(free, l -> !l.isEmpty(),
                              l -> release(l, next, waits))
                .mapToInt(List::size).sum() == n;
    }

    // ---- the same two helpers in 207, 210, 802 and 1203 ----
    // edges [from, to], grouped as from -> every to
    static Map<Integer, List<Integer>> adj(Stream<int[]> es) {
        return es.collect(Collectors.groupingBy(e -> e[0],
                Collectors.mapping(e -> e[1],
                        Collectors.toList())));
    }

    // Kahn's next frontier. The decrement is the state: a node
    // joins exactly when its last incoming edge is used up.
    static List<Integer> release(List<Integer> level,
            Map<Integer, List<Integer>> next, int[] waits) {
        return level.stream()
                .flatMap(u -> next.getOrDefault(u, List.of())
                        .stream())
                .filter(v -> --waits[v] == 0)
                .toList();
    }
}
LC 210 · Course Schedule II · stateful
class Solution {
    public int[] findOrder(int n, int[][] prerequisites) {
        Map<Integer, List<Integer>> next = adj(Arrays
                .stream(prerequisites)
                .map(e -> new int[]{ e[1], e[0] }));
        int[] waits = Arrays.stream(prerequisites).collect(
                () -> new int[n], (w, e) -> w[e[0]]++,
                (a, b) -> Arrays.setAll(a, i -> a[i] + b[i]));
        List<Integer> free = IntStream.range(0, n)
                .filter(c -> waits[c] == 0).boxed().toList();
        // The frontiers of 207, flattened: a topological order.
        int[] order = Stream
                .iterate(free, l -> !l.isEmpty(),
                         l -> release(l, next, waits))
                .flatMap(List::stream)
                .mapToInt(Integer::intValue).toArray();
        // shorter than n: a cycle kept some course waiting
        return order.length == n ? order : new int[0];
    }

    // ---- the same two helpers in 207, 210, 802 and 1203 ----
    // edges [from, to], grouped as from -> every to

        return es.collect(Collectors.groupingBy(e -> e[0],

LC 802 · Find Eventual Safe States · stateful
class Solution {
    public List<Integer> eventualSafeNodes(int[][] graph) {
        int n = graph.length;
        // Reverse every edge u -> v, so that safety can flow
        // backwards from the terminal nodes.
        Map<Integer, List<Integer>> into = adj(IntStream
                .range(0, n).boxed()
                .flatMap(u -> Arrays.stream(graph[u])
                        .mapToObj(v -> new int[]{ v, u })));
        int[] out = Arrays.stream(graph).mapToInt(a -> a.length)
                .toArray();
        // Kahn from the terminals: a node is released, so safe,
        // once every edge out of it leads somewhere safe.
        List<Integer> terminal = IntStream.range(0, n)
                .filter(u -> out[u] == 0).boxed().toList();
        return Stream.iterate(terminal, l -> !l.isEmpty(),
                              l -> release(l, into, out))
                .flatMap(List::stream)
                .sorted()
                .toList();
    }

    // ---- the same two helpers in 207, 210, 802 and 1203 ----
    // edges [from, to], grouped as from -> every to
    static Map<Integer, List<Integer>> adj(Stream<int[]> es) {
        return es.collect(Collectors.groupingBy(e -> e[0],
                Collectors.mapping(e -> e[1],
                        Collectors.toList())));
    }

    // Kahn's next frontier. The decrement is the state: a node
    // joins exactly when its last incoming edge is used up.
    static List<Integer> release(List<Integer> level,
            Map<Integer, List<Integer>> next, int[] waits) {
        return level.stream()
                .flatMap(u -> next.getOrDefault(u, List.of())
                        .stream())
                .filter(v -> --waits[v] == 0)
                .toList();
    }
}
LC 2115 · Find All Possible Recipes from Given Supplies · stateful
class Solution {
    public List<String> findAllRecipes(String[] recipes,
            List<List<String>> ingredients, String[] supplies) {
        // ingredient name -> the recipes that need it
        Map<String, List<Integer>> usedBy = IntStream
                .range(0, recipes.length).boxed()
                .flatMap(i -> ingredients.get(i).stream()
                        .map(item -> Map.entry(item, i)))
                .collect(Collectors.groupingBy(e -> e.getKey(),
                        Collectors.mapping(e -> e.getValue(),
                                Collectors.toList())));
        int[] missing = ingredients.stream()
                .mapToInt(List::size).toArray();
        // Kahn over names: supplies are the first frontier, and
        // a recipe is released along with its last ingredient.
        List<String> start = List.of(supplies);
        return Stream.iterate(start, l -> !l.isEmpty(),
                l -> unlock(l, recipes, usedBy, missing))
                .skip(1)                     // the supplies
                .flatMap(List::stream)
                .toList();
    }

    // The recipes whose last missing ingredient is in level.
    // The decrement is the stateful line.
    static List<String> unlock(List<String> level,
            String[] recipes, Map<String, List<Integer>> usedBy,
            int[] missing) {
        return level.stream()
                .flatMap(s -> usedBy.getOrDefault(s, List.of())
                        .stream())
                .filter(i -> --missing[i] == 0)
                .map(i -> recipes[i])
                .toList();
    }
}
LC 1462 · Course Schedule IV · clean
class Solution {
    // after k steps, reach[i] holds every course reachable
    // from i through intermediate courses < k (Floyd-Warshall)
    record Closure(int k, BitSet[] reach) {
        Closure next() {
            BitSet[] r = Arrays.stream(reach).map(row -> {
                BitSet copy = (BitSet) row.clone();
                if (row.get(k)) copy.or(reach[k]);
                return copy;
            }).toArray(BitSet[]::new);
            return new Closure(k + 1, r);
        }
    }

    public List<Boolean> checkIfPrerequisite(int n,
            int[][] prerequisites, int[][] queries) {
        // Direct edges; BitSet::or is a real combiner, so this
        // collect is parallel-safe.
        BitSet[] direct = IntStream.range(0, n)
                .mapToObj(u -> Arrays.stream(prerequisites)
                        .filter(e -> e[0] == u)
                        .mapToInt(e -> e[1])
                        .collect(BitSet::new, BitSet::set,
                                 BitSet::or))
                .toArray(BitSet[]::new);
        // n pure steps, each allowing paths through one more
        // course. No traversal, so no visited marks anywhere.
        BitSet[] reach = Stream.iterate(new Closure(0, direct),
                                        Closure::next)
                .skip(n).findFirst().get().reach();
        return Arrays.stream(queries)
                .map(q -> reach[q[0]].get(q[1]))
                .toList();
    }
}
LC 1203 · Sort Items by Groups Respecting Dependencies · stateful
class Solution {
    public int[] sortItems(int n, int m, int[] group,
                           List<List<Integer>> beforeItems) {
        // An item without a group gets a group of its own.
        int[] g = IntStream.range(0, n)
                .map(i -> group[i] >= 0 ? group[i] : m + i)
                .toArray();
        List<int[]> items = IntStream.range(0, n).boxed()
                .flatMap(i -> beforeItems.get(i).stream()
                        .map(b -> new int[]{ b, i }))
                .toList();
        // group edges: the item edges that cross between groups
        List<int[]> groups = items.stream()
                .filter(e -> g[e[0]] != g[e[1]])
                .map(e -> new int[]{ g[e[0]], g[e[1]] })
                .toList();
        List<Integer> itemOrder = topo(n, items);
        List<Integer> groupOrder = topo(m + n, groups);
        if (itemOrder.size() < n || groupOrder.size() < m + n)
            return new int[0];
        // Items keep their global order inside each group, and
        // the groups come out in theirs.
        Map<Integer, List<Integer>> byGroup = itemOrder.stream()
                .collect(Collectors.groupingBy(i -> g[i]));
        return groupOrder.stream()
                .flatMap(x -> byGroup.getOrDefault(x, List.of())
                        .stream())
                .mapToInt(Integer::intValue).toArray();
    }

    // Kahn by frontiers over [from, to] edges
    static List<Integer> topo(int n, List<int[]> edges) {
        Map<Integer, List<Integer>> next = adj(edges.stream());
        int[] waits = edges.stream().collect(() -> new int[n],
                (w, e) -> w[e[1]]++,
                (a, b) -> Arrays.setAll(a, i -> a[i] + b[i]));
        List<Integer> free = IntStream.range(0, n)
                .filter(v -> waits[v] == 0).boxed().toList();
        return Stream.iterate(free, l -> !l.isEmpty(),
                              l -> release(l, next, waits))
                .flatMap(List::stream)
                .toList();
    }

    // ---- the same two helpers in 207, 210, 802 and 1203 ----
    // edges [from, to], grouped as from -> every to
    static Map<Integer, List<Integer>> adj(Stream<int[]> es) {
        return es.collect(Collectors.groupingBy(e -> e[0],
                Collectors.mapping(e -> e[1],
                        Collectors.toList())));
    }

    // Kahn's next frontier. The decrement is the state: a node
    // joins exactly when its last incoming edge is used up.
    static List<Integer> release(List<Integer> level,
            Map<Integer, List<Integer>> next, int[] waits) {
        return level.stream()
                .flatMap(u -> next.getOrDefault(u, List.of())
                        .stream())
                .filter(v -> --waits[v] == 0)
                .toList();
    }
}
The decrement carries the algorithmfilter(v -> --waits[v] == 0) admits a node exactly when its last prerequisite leaves. It is correct only because each edge is seen once, in order, and that makes it stateful. It is the one line in these solutions that must never run in parallel, and keeping it inside a single helper, release, makes that easy to see.
BitSet collectscollect(BitSet::new, BitSet::set, BitSet::or) is a mutable reduction whose combiner is a real union, so building a set of indices this way is safe in parallel. 1462's closure is then a fold over fresh BitSet rows, with no visited marks anywhere. At n ≤ 100 that is 100 steps of cheap bit-level ORs.
