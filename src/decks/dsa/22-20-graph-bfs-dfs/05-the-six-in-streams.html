BFS levels are a Stream.iterate; the visited mark is the one stateful lambda

A grid search needs a visited set, and a stream stage cannot own one. So every solution here keeps exactly one stateful lambda, claim, which tests a cell and marks it in the same step. Everything around it is structure: Stream.iterate produces the BFS levels, and mapToLong turns each cell into the size of the island it starts. Nothing recurses, so a 300 × 300 snake-shaped island costs no stack.

LC 733 · Flood Fill · stateful
class Solution {
    public int[][] floodFill(int[][] image, int sr, int sc,
                             int color) {
        int m = image.length, n = image[0].length;
        int old = image[sr][sc];
        if (old == color) return image;
        // Painting a cell is what marks it as visited.
        IntPredicate paint = x -> {
            if (image[x / n][x % n] != old) return false;
            image[x / n][x % n] = color;
            return true;
        };
        int start = sr * n + sc;
        paint.test(start);
        flood(start, m, n, paint).count();   // runs the walk
        return image;
    }

    // ---- the same two helpers in 733, 200, 695 and 1254 ----
    // The 4 neighbours of cell x = r * n + c, inside the grid.
    static IntStream around(int x, int m, int n) {
        int c = x % n;
        int left = c > 0 ? x - 1 : -1;
        int right = c < n - 1 ? x + 1 : -1;
        return IntStream.of(x - n, x + n, left, right)
                .filter(y -> y >= 0 && y < m * n);
    }

    // BFS level by level from start. claim tests AND marks a
    // cell: the stateful lambda. Returns every cell reached.
    static Stream<Integer> flood(int start, int m, int n,
                                 IntPredicate claim) {
        return Stream.iterate(List.of(start), l -> !l.isEmpty(),
                        l -> l.stream()
                                .flatMap(x -> around(x, m, n)
                                        .filter(claim).boxed())
                                .toList())
                .flatMap(List::stream);
    }
}
LC 200 · Number of Islands · stateful
class Solution {
    public int numIslands(char[][] grid) {
        int m = grid.length, n = grid[0].length;
        // Claiming a cell sinks it; that write is the visited
        // set, and the one stateful lambda here.
        IntPredicate claim = x -> {
            if (grid[x / n][x % n] != '1') return false;
            grid[x / n][x % n] = '0';
            return true;
        };
        // Each cell maps to the size of the island it starts:
        // 0 unless it is land nobody has claimed yet.
        return (int) IntStream.range(0, m * n)
                .mapToLong(x -> claim.test(x)
                        ? flood(x, m, n, claim).count() : 0)
                .filter(size -> size > 0)
                .count();
    }

    // ---- the same two helpers in 733, 200, 695 and 1254 ----
    // The 4 neighbours of cell x = r * n + c, inside the grid.
    static IntStream around(int x, int m, int n) {
        int c = x % n;
        int left = c > 0 ? x - 1 : -1;
        int right = c < n - 1 ? x + 1 : -1;
        return IntStream.of(x - n, x + n, left, right)
                .filter(y -> y >= 0 && y < m * n);
    }

    // BFS level by level from start. claim tests AND marks a
    // cell: the stateful lambda. Returns every cell reached.
    static Stream<Integer> flood(int start, int m, int n,
                                 IntPredicate claim) {
        return Stream.iterate(List.of(start), l -> !l.isEmpty(),
                        l -> l.stream()
                                .flatMap(x -> around(x, m, n)
                                        .filter(claim).boxed())
                                .toList())
                .flatMap(List::stream);
    }
}
LC 695 · Max Area of Island · stateful
class Solution {
    public int maxAreaOfIsland(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        IntPredicate claim = x -> {
            if (grid[x / n][x % n] != 1) return false;
            grid[x / n][x % n] = 0;
            return true;
        };
        // 200 with max() instead of count(): each cell maps
        // to the area of the island it starts.
        return (int) IntStream.range(0, m * n)

                .max().orElse(0);

LC 1254 · Number of Closed Islands · stateful
class Solution {
    public int closedIsland(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        // Here land is 0; claiming turns it into water (1).
        IntPredicate claim = x -> {
            if (grid[x / n][x % n] != 0) return false;
            grid[x / n][x % n] = 1;
            return true;
        };
        IntToLongFunction island = x -> claim.test(x)
                ? flood(x, m, n, claim).count() : 0;
        IntPredicate edge = x -> x / n == 0 || x / n == m - 1
                || x % n == 0 || x % n == n - 1;
        // First sink every island that touches the edge, then
        // count the islands that remain: those are closed.
        IntStream.range(0, m * n).filter(edge).mapToLong(island)
                .sum();
        return (int) IntStream.range(0, m * n).mapToLong(island)
                .filter(size -> size > 0)
                .count();
    }

    // ---- the same two helpers in 733, 200, 695 and 1254 ----
    // The 4 neighbours of cell x = r * n + c, inside the grid.
    static IntStream around(int x, int m, int n) {
        int c = x % n;
        int left = c > 0 ? x - 1 : -1;
        int right = c < n - 1 ? x + 1 : -1;
        return IntStream.of(x - n, x + n, left, right)
                .filter(y -> y >= 0 && y < m * n);
    }

    // BFS level by level from start. claim tests AND marks a
    // cell: the stateful lambda. Returns every cell reached.

                                 IntPredicate claim) {

LC 994 · Rotting Oranges · stateful
class Solution {
    public int orangesRotting(int[][] grid) {
        int m = grid.length, n = grid[0].length;
        // fresh -> rotten; the write also marks it visited
        IntPredicate rot = x -> {
            if (grid[x / n][x % n] != 1) return false;
            grid[x / n][x % n] = 2;
            return true;
        };
        List<Integer> rotten = IntStream.range(0, m * n)
                .filter(x -> grid[x / n][x % n] == 2)
                .boxed().toList();
        // Multi-source BFS: every level after the first is one
        // minute of spreading.
        long levels = Stream.iterate(rotten, l -> !l.isEmpty(),
                        l -> l.stream()
                                .flatMap(x -> around(x, m, n)
                                        .filter(rot).boxed())
                                .toList())
                .count();
        boolean fresh = Arrays.stream(grid)
                .flatMapToInt(Arrays::stream)
                .anyMatch(v -> v == 1);
        return fresh ? -1 : (int) Math.max(0, levels - 1);
    }

    // The 4 neighbours of cell x = r * n + c, inside the grid.
    static IntStream around(int x, int m, int n) {
        int c = x % n;
        int left = c > 0 ? x - 1 : -1;
        int right = c < n - 1 ? x + 1 : -1;
        return IntStream.of(x - n, x + n, left, right)
                .filter(y -> y >= 0 && y < m * n);
    }
}
LC 1091 · Shortest Path in Binary Matrix · stateful
class Solution {
    public int shortestPathBinaryMatrix(int[][] grid) {
        int n = grid.length, target = n * n - 1;
        if (grid[0][0] != 0 || grid[n - 1][n - 1] != 0)
            return -1;
        IntPredicate claim = x -> {       // open -> visited
            if (grid[x / n][x % n] != 0) return false;
            grid[x / n][x % n] = 1;
            return true;
        };
        claim.test(0);
        // Level k holds the cells a path of k + 1 cells reaches
        // first. Count the levels before the target's.
        long before = Stream
                .iterate(List.of(0), l -> !l.isEmpty(),
                        l -> l.stream()
                                .flatMap(x -> around8(x, n)
                                        .filter(claim).boxed())
                                .toList())
                .takeWhile(l -> !l.contains(target))
                .count();
        return grid[n - 1][n - 1] == 1 ? (int) before + 1 : -1;
    }

    // the 8 neighbours of cell x = r * n + c, inside the grid
    static IntStream around8(int x, int n) {
        int r = x / n, c = x % n;
        return IntStream.rangeClosed(r - 1, r + 1)
                .filter(y -> y >= 0 && y < n)
                .flatMap(y -> IntStream
                        .rangeClosed(c - 1, c + 1)
                        .filter(z -> z >= 0 && z < n)
                        .map(z -> y * n + z))
                .filter(y -> y != x);
    }
}
Why the mark goes inside the filterclaim must test and mark in one step. Two cells on the same level can share a neighbour. If the mark came later, that neighbour would enter the next level twice. The loop marks a cell when it is enqueued, for the same reason. Here that rule becomes "the filter that admits a cell is the one that marks it", which is correct only because the stream runs sequentially.
One mapping, three problems200, 695 and 1254 are the same pipeline with a different ending. Each cell maps to the size of the island it starts: count the sizes above zero, take the max, or sink the edge islands first. Spotting that shared shape is the real lesson of the pattern, and the stream makes it visible because the ending is the only line that changes.
