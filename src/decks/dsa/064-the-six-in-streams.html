A fixed array needs a sparse table; values still being written need the deque

Split the six by what the window reads. When the array is complete before any query (239, 1438, 1499), the max over any range comes from a sparse table. Each level doubles the block length and is a pure function of the level before it, so Stream.iterate builds it and every query is O(1). When the window reads DP values the stream is still writing (1696, 1425), or the search has to happen inside the window as it moves (862), only the monotonic deque works. A deque inside a lambda is the loop.

LC 239 · Sliding Window Maximum · clean
class Solution {
    public int[] maxSlidingWindow(int[] nums, int k) {
        // The array is fixed before any query, so every window
        // max is one O(1) lookup in a sparse table.
        IntBinaryOperator max = rangeOf(nums, Math::max);
        return IntStream.rangeClosed(0, nums.length - k)
                .map(i -> max.applyAsInt(i, i + k))
                .toArray();
    }

    // ---- the same sparse table in 239, 1438 and 1499 ----
    // v[i] = op over a[i, i + len); doubling len is pure
    record Level(int len, int[] v) {
        Level doubled(IntBinaryOperator op) {
            int[] w = IntStream.range(0, v.length - len)
                    .map(i -> op.applyAsInt(v[i], v[i + len]))
                    .toArray();
            return new Level(2 * len, w);
        }
    }

    // op over [lo, hi) in O(1): two overlapping blocks of 2^j
    static IntBinaryOperator rangeOf(int[] a,
                                     IntBinaryOperator op) {
        List<int[]> t = Stream.iterate(new Level(1, a),
                                       l -> l.doubled(op))
                .takeWhile(l -> l.len() <= a.length)
                .map(Level::v)
                .toList();
        return (lo, hi) -> {
            int j = 31 - Integer.numberOfLeadingZeros(hi - lo);
            int[] v = t.get(j);
            return op.applyAsInt(v[lo], v[hi - (1 << j)]);
        };
    }
}
LC 1438 · Longest Continuous Subarray With Absolute Diff ≤ Limit · clean
class Solution {
    // window [l, r) and the best length seen so far
    record Win(int l, int r, int best) {
        Win next(boolean fits) {
            return fits
                    ? new Win(l, r + 1, Math.max(best, r - l))
                    : new Win(l + 1, r, best);
        }
    }

    public int longestSubarray(int[] nums, int limit) {
        int n = nums.length;
        IntBinaryOperator max = rangeOf(nums, Math::max);
        IntBinaryOperator min = rangeOf(nums, Math::min);
        // A state machine, as in the sliding-window slides:
        // grow r while [l, r) fits, else drop l. The two tables
        // make each step O(1), so no deque is needed.
        return Stream.iterate(new Win(0, 1, 0), w -> w.next(
                    max.applyAsInt(w.l(), w.r())
                    - min.applyAsInt(w.l(), w.r()) <= limit))
                .filter(w -> w.r() > n)
                .findFirst().get().best();
    }

    // ---- the same sparse table in 239, 1438 and 1499 ----
    // v[i] = op over a[i, i + len); doubling len is pure
    record Level(int len, int[] v) {
        Level doubled(IntBinaryOperator op) {
            int[] w = IntStream.range(0, v.length - len)
                    .map(i -> op.applyAsInt(v[i], v[i + len]))
                    .toArray();
            return new Level(2 * len, w);
        }
    }

    // op over [lo, hi) in O(1): two overlapping blocks of 2^j
    static IntBinaryOperator rangeOf(int[] a,
                                     IntBinaryOperator op) {
        List<int[]> t = Stream.iterate(new Level(1, a),
                                       l -> l.doubled(op))
                .takeWhile(l -> l.len() <= a.length)
                .map(Level::v)
                .toList();
        return (lo, hi) -> {
            int j = 31 - Integer.numberOfLeadingZeros(hi - lo);
            int[] v = t.get(j);
            return op.applyAsInt(v[lo], v[hi - (1 << j)]);
        };
    }
}
LC 1696 · Jump Game VI · forced
class Solution {
    public int maxResult(int[] nums, int k) {
        int n = nums.length;
        int[] dp = new int[n];
        Deque<Integer> dq = new ArrayDeque<>(); // dp decreasing
        // dp is written while the stream runs, so no table
        // built up front can answer the window max. Only the
        // deque can, and a deque inside a lambda is the loop.
        IntStream.range(0, n).forEach(i -> {
            if (!dq.isEmpty() && dq.peekFirst() < i - k)
                dq.pollFirst();
            dp[i] = nums[i] + (i == 0 ? 0 : dp[dq.peekFirst()]);
            while (!dq.isEmpty() && dp[dq.peekLast()] <= dp[i])
                dq.pollLast();
            dq.offerLast(i);
        });
        return dp[n - 1];
    }
}
LC 1425 · Constrained Subsequence Sum · forced
class Solution {
    public int constrainedSubsetSum(int[] nums, int k) {
        int n = nums.length;
        int[] dp = new int[n];
        Deque<Integer> dq = new ArrayDeque<>(); // dp decreasing
        // 1696 with an opt-out: a negative best becomes 0,
        // because a subsequence may start fresh at i.
        return IntStream.range(0, n).map(i -> {
            if (!dq.isEmpty() && dq.peekFirst() < i - k)
                dq.pollFirst();
            int best = dq.isEmpty() ? 0 : dp[dq.peekFirst()];
            dp[i] = nums[i] + Math.max(0, best);
            while (!dq.isEmpty() && dp[dq.peekLast()] <= dp[i])
                dq.pollLast();
            dq.offerLast(i);
            return dp[i];
        }).max().getAsInt();
    }
}
LC 862 · Shortest Subarray with Sum at Least K · forced
class Solution {
    public int shortestSubarray(int[] nums, int k) {
        int n = nums.length;
        long[] run = {0};
        long[] p = LongStream.concat(LongStream.of(0),
                        Arrays.stream(nums).asLongStream()
                                .map(v -> run[0] += v))
                .toArray();
        Deque<Integer> dq = new ArrayDeque<>();  // p increasing
        // Front: every start that already reaches k is done,
        // as it can only get longer. Back: a start no lower
        // than p[j] is never useful again.
        int best = IntStream.rangeClosed(0, n).map(j -> {
            int len = Integer.MAX_VALUE;
            while (!dq.isEmpty()
                    && p[j] - p[dq.peekFirst()] >= k)
                len = Math.min(len, j - dq.pollFirst());
            while (!dq.isEmpty() && p[dq.peekLast()] >= p[j])
                dq.pollLast();
            dq.offerLast(j);
            return len;
        }).min().getAsInt();
        return best == Integer.MAX_VALUE ? -1 : best;
    }
}
LC 1499 · Max Value of Equation · clean
class Solution {
    public int findMaxValueOfEquation(int[][] points, int k) {
        int[] x = Arrays.stream(points)
                .mapToInt(p -> p[0]).toArray();
        int[] yx = Arrays.stream(points)
                .mapToInt(p -> p[1] - p[0]).toArray();
        IntBinaryOperator best = rangeOf(yx, Math::max);
        // y_i + y_j + x_j - x_i = (x_j + y_j) + (y_i - x_i).
        // The points within k left of j are one range, as x is
        // sorted: binary-search its start, then query its max.
        return IntStream.range(1, points.length).map(j -> {
            int f = Arrays.binarySearch(x, 0, j, x[j] - k);
            int lo = f >= 0 ? f : -f - 1;
            if (lo == j) return Integer.MIN_VALUE;
            return x[j] + points[j][1] + best.applyAsInt(lo, j);
        }).max().getAsInt();
    }

    // ---- the same sparse table in 239, 1438 and 1499 ----
    // v[i] = op over a[i, i + len); doubling len is pure
    record Level(int len, int[] v) {
        Level doubled(IntBinaryOperator op) {
            int[] w = IntStream.range(0, v.length - len)
                    .map(i -> op.applyAsInt(v[i], v[i + len]))
                    .toArray();
            return new Level(2 * len, w);
        }
    }

    // op over [lo, hi) in O(1): two overlapping blocks of 2^j
    static IntBinaryOperator rangeOf(int[] a,
                                     IntBinaryOperator op) {
        List<int[]> t = Stream.iterate(new Level(1, a),
                                       l -> l.doubled(op))
                .takeWhile(l -> l.len() <= a.length)
                .map(Level::v)
                .toList();
        return (lo, hi) -> {
            int j = 31 - Integer.numberOfLeadingZeros(hi - lo);
            int[] v = t.get(j);
            return op.applyAsInt(v[lo], v[hi - (1 << j)]);
        };
    }
}
The idea worth keepingDoubling is how a stream builds what would otherwise need a mutable window. Level j + 1 at i is op(level j at i, level j at i + 2ʲ), so Stream.iterate(level0, Level::doubled).takeWhile(…) is the whole sparse table, and it is pure. It costs a log factor in time and memory, which these constraints allow. In exchange, any range max or min becomes one lookup, with no deque in sight.
Where streams stop1696 and 1425 read DP values that the same pass is still writing, so no table can be built up front. The deque in a forEach is the loop with extra ceremony. In an interview, write the loop and say why a pipeline cannot help: the window's contents are produced by the same pass that queries them.
