A constraint is a filter on the branches; state that fits in a value needs no undo

Each problem here is the basic backtracking flatMap with a filter deciding which branches exist: a palindrome test, a skip over equal siblings, a check that a square is safe. 79 adds a visited set. On a board of at most 6 × 6 that set fits in one long, so it travels down the recursion as a value and nothing is ever unmarked.

LC 17 · Letter Combinations of a Phone Number · clean
class Solution {
    static final String[] KEYS = { "", "", "abc", "def", "ghi",
            "jkl", "mno", "pqrs", "tuv", "wxyz" };

    public List<String> letterCombinations(String digits) {
        if (digits.isEmpty()) return List.of();
        return words(digits, 0).toList();
    }

    // Each letter of digit i, then every word for the rest.
    static Stream<String> words(String d, int i) {
        if (i == d.length()) return Stream.of("");
        return KEYS[d.charAt(i) - '0'].chars()
                .mapToObj(c -> String.valueOf((char) c))
                .flatMap(c -> words(d, i + 1)
                        .map(rest -> c + rest));
    }
}
LC 40 · Combination Sum II · clean
class Solution {
    public List<List<Integer>> combinationSum2(int[] candidates,
                                               int target) {
        int[] c = candidates.clone();
        Arrays.sort(c);
        return sums(c, 0, target).toList();
    }

    static Stream<List<Integer>> sums(int[] c, int from,
                                      int left) {
        if (left == 0) return Stream.of(List.of());
        // Sorted: takeWhile ends the level at the first value
        // too big, and the filter skips a repeat of a sibling.
        return IntStream.range(from, c.length)
                .takeWhile(i -> c[i] <= left)
                .filter(i -> i == from || c[i] != c[i - 1])
                .boxed()
                .flatMap(i -> sums(c, i + 1, left - c[i])
                        .map(rest -> prepend(c[i], rest)));
    }

    static List<Integer> prepend(int x, List<Integer> rest) {
        return Stream.concat(Stream.of(x), rest.stream())
                .toList();
    }
}
LC 131 · Palindrome Partitioning · clean
class Solution {
    public List<List<String>> partition(String s) {
        return parts(s, 0).toList();
    }

    // a palindromic first piece, then any partition of the rest
    static Stream<List<String>> parts(String s, int from) {
        if (from == s.length()) return Stream.of(List.of());
        return IntStream.rangeClosed(from + 1, s.length())
                .filter(end -> isPal(s, from, end))
                .boxed()
                .flatMap(end -> parts(s, end).map(rest ->
                        prepend(s.substring(from, end), rest)));
    }

    static boolean isPal(String s, int lo, int hi) {  // [lo,hi)
        return IntStream.range(0, (hi - lo) / 2).allMatch(
                k -> s.charAt(lo + k) == s.charAt(hi - 1 - k));
    }

    static List<String> prepend(String x, List<String> rest) {
        return Stream.concat(Stream.of(x), rest.stream())
                .toList();
    }
}
LC 79 · Word Search · clean
class Solution {
    public boolean exist(char[][] board, String word) {
        // Cheap prune first: the board must hold every letter
        // the word needs, as many times as it needs it.
        Map<Integer, Long> have = count(Arrays.stream(board)
                .flatMapToInt(row -> new String(row).chars()));
        Map<Integer, Long> need = count(word.chars());
        if (need.keySet().stream().anyMatch(
                ch -> have.getOrDefault(ch, 0L) < need.get(ch)))
            return false;
        int cells = board.length * board[0].length;
        return IntStream.range(0, cells)
                .anyMatch(at -> from(board, word, 0, at, 0L));
    }

    static Map<Integer, Long> count(IntStream s) {
        return s.boxed().collect(Collectors.groupingBy(
                c -> c, Collectors.counting()));
    }

    // Can word[i..] start at cell without reusing a cell in
    // seen? At most 36 cells, so seen is a long: passed down
    // as a value, never marked and unmarked.
    static boolean from(char[][] b, String w, int i, int cell,
                        long seen) {
        int n = b[0].length, r = cell / n, c = cell % n;
        if (b[r][c] != w.charAt(i) || (seen >> cell & 1) == 1)
            return false;
        if (i == w.length() - 1) return true;
        long now = seen | 1L << cell;
        int[][] next = { { r - 1, c }, { r + 1, c },
                         { r, c - 1 }, { r, c + 1 } };
        return Arrays.stream(next)
                .filter(p -> p[0] >= 0 && p[0] < b.length
                          && p[1] >= 0 && p[1] < n)
                .anyMatch(p -> from(b, w, i + 1,
                                    p[0] * n + p[1], now));
    }
}
LC 51 · N-Queens · clean
class Solution {
    public List<List<String>> solveNQueens(int n) {
        return place(n, List.of())
                .map(cols -> cols.stream()
                        .map(c -> ".".repeat(c) + "Q"
                                + ".".repeat(n - 1 - c))
                        .toList())
                .toList();
    }

    // cols.get(r) = the queen's column in row r. Each row may
    // only use a column that no earlier queen attacks.
    static Stream<List<Integer>> place(int n,
                                       List<Integer> cols) {
        int r = cols.size();
        if (r == n) return Stream.of(cols);
        IntPredicate free = c -> IntStream.range(0, r)
                .noneMatch(q -> cols.get(q) == c
                        || Math.abs(cols.get(q) - c) == r - q);
        return IntStream.range(0, n).filter(free).boxed()
                .flatMap(c -> place(n, Stream.concat(
                        cols.stream(), Stream.of(c)).toList()));
    }
}
LC 52 · N-Queens II · clean
class Solution {
    public int totalNQueens(int n) {
        // 51's search, counted instead of drawn.
        return (int) place(n, List.of()).count();
    }

    // cols.get(r) = the queen's column in row r. Each row may
    // only use a column that no earlier queen attacks.
    static Stream<List<Integer>> place(int n,
                                       List<Integer> cols) {
        int r = cols.size();
        if (r == n) return Stream.of(cols);
        IntPredicate free = c -> IntStream.range(0, r)
                .noneMatch(q -> cols.get(q) == c
                        || Math.abs(cols.get(q) - c) == r - q);
        return IntStream.range(0, n).filter(free).boxed()
                .flatMap(c -> place(n, Stream.concat(
                        cols.stream(), Stream.of(c)).toList()));
    }
}
A visited set that is a value79's board has at most 36 cells, so "visited" fits in a long. seen | 1L << cell makes a new value for the next step instead of changing a shared one. That turns the one problem here with genuine backtracking state into a pure recursion. On a grid bigger than 64 cells the trick no longer fits, and a mutable visited array (stateful) comes back.
Where the pruning livestakeWhile on sorted candidates (40) ends a level the moment a value overshoots, which makes it the stream version of break. filter only skips. 79's letter-count check is the other kind of prune: one groupingBy rejects a board that lacks a letter before the search starts. Without it, a 6 × 6 board of A's took 0.65 s to reject a word ending in B.
