The search is one helper; the feasibility test is a pipeline

Binary search on the answer has two parts: a search over candidate values, and a yes-or-no test for one candidate. The search is the sorted-data pattern's firstTrue helper, unchanged. The test is nearly always a sum or a count over the input, which makes it a single pipeline. Even the greedy packing in 1011 and 410 becomes a pure fold over a three-field record.

LC 875 · Koko Eating Bananas · clean
class Solution {
    public int minEatingSpeed(int[] piles, int h) {
        int max = Arrays.stream(piles).max().getAsInt();
        // Hours only shrink as the speed grows, so "fits in h"
        // is monotone: search for the first speed that fits.
        return firstTrue(1, max, k -> Arrays.stream(piles)
                .mapToLong(p -> (p + k - 1) / k)  // ceil(p/k)
                .sum() <= h);
    }

    // ---- the same helper as in the sorted-data search ----
    // [lo, hi) halves towards the first index where ok holds
    record Range(int lo, int hi) {
        Range halve(IntPredicate ok) {
            int mid = (lo + hi) >>> 1;
            return ok.test(mid) ? new Range(lo, mid)
                                : new Range(mid + 1, hi);
        }
    }

    // First value in [lo, hi) where ok holds, else hi.
    static int firstTrue(int lo, int hi, IntPredicate ok) {
        return Stream.iterate(new Range(lo, hi),
                              r -> r.halve(ok))
                .filter(r -> r.lo() == r.hi())
                .findFirst().get().lo();
    }
}
LC 1011 · Capacity To Ship Packages Within D Days · clean
class Solution {
    public int shipWithinDays(int[] weights, int days) {
        int max = Arrays.stream(weights).max().getAsInt();
        int sum = Arrays.stream(weights).sum();
        // The least capacity whose greedy packing fits in days.
        return firstTrue(max, sum,
                         cap -> need(weights, cap) <= days);
    }

    // Greedy packing as a fold: one package per step, carrying
    // the days used so far and the load on the current day.
    record Pack(int i, int days, int load) {
        Pack add(int w, int cap) {
            return load + w > cap
                    ? new Pack(i + 1, days + 1, w)
                    : new Pack(i + 1, days, load + w);
        }
    }

    static int need(int[] w, int cap) {
        return Stream.iterate(new Pack(0, 1, 0),
                              p -> p.add(w[p.i()], cap))
                .skip(w.length)   // exactly n steps
                .findFirst().get().days();
    }

    // ---- the same helper as in the sorted-data search ----
    // [lo, hi) halves towards the first index where ok holds
    record Range(int lo, int hi) {
        Range halve(IntPredicate ok) {
            int mid = (lo + hi) >>> 1;
            return ok.test(mid) ? new Range(lo, mid)
                                : new Range(mid + 1, hi);
        }
    }

    // First value in [lo, hi) where ok holds, else hi.
    static int firstTrue(int lo, int hi, IntPredicate ok) {
        return Stream.iterate(new Range(lo, hi),
                              r -> r.halve(ok))

LC 1482 · Minimum Number of Days to Make m Bouquets · clean
class Solution {
    public int minDays(int[] bloomDay, int m, int k) {
        if ((long) m * k > bloomDay.length) return -1;
        int max = Arrays.stream(bloomDay).max().getAsInt();
        return firstTrue(1, max,
                day -> bouquets(bloomDay, day, k) >= m);
    }

    // Unbloomed flowers cut the row into runs, and a run of
    // length len makes len / k bouquets. Read the runs off the
    // gaps between consecutive cuts: nothing is carried.
    static long bouquets(int[] bloom, int day, int k) {
        int n = bloom.length;
        int[] cut = IntStream.concat(IntStream.of(-1),
                IntStream.concat(IntStream.range(0, n)
                                .filter(i -> bloom[i] > day),
                        IntStream.of(n)))
                .toArray();
        return IntStream.range(1, cut.length)
                .mapToLong(j -> (cut[j] - cut[j - 1] - 1) / k)
                .sum();
    }

    // ---- the same helper as in the sorted-data search ----
    // [lo, hi) halves towards the first index where ok holds
    record Range(int lo, int hi) {
        Range halve(IntPredicate ok) {
            int mid = (lo + hi) >>> 1;
            return ok.test(mid) ? new Range(lo, mid)
                                : new Range(mid + 1, hi);
        }
    }

    // First value in [lo, hi) where ok holds, else hi.
    static int firstTrue(int lo, int hi, IntPredicate ok) {
        return Stream.iterate(new Range(lo, hi),
                              r -> r.halve(ok))
                .filter(r -> r.lo() == r.hi())
                .findFirst().get().lo();
    }
}
LC 410 · Split Array Largest Sum · clean
class Solution {
    public int splitArray(int[] nums, int k) {
        int max = Arrays.stream(nums).max().getAsInt();
        int sum = Arrays.stream(nums).sum();
        // 1011 in disguise: the least cap on each piece's sum
        // whose greedy split needs at most k pieces.
        return firstTrue(max, sum,
                         cap -> pieces(nums, cap) <= k);
    }

    // Greedy split as a fold: pieces so far, current load.
    record Pack(int i, int pieces, int load) {
        Pack add(int x, int cap) {
            return load + x > cap
                    ? new Pack(i + 1, pieces + 1, x)
                    : new Pack(i + 1, pieces, load + x);
        }
    }

    static int pieces(int[] a, int cap) {
        return Stream.iterate(new Pack(0, 1, 0),
                              p -> p.add(a[p.i()], cap))
                .skip(a.length)
                .findFirst().get().pieces();
    }

    // ---- the same helper as in the sorted-data search ----
    // [lo, hi) halves towards the first index where ok holds

        Range halve(IntPredicate ok) {
            int mid = (lo + hi) >>> 1;
            return ok.test(mid) ? new Range(lo, mid)
                                : new Range(mid + 1, hi);
        }
    }

LC 774 · Minimize Max Distance to Gas Station · clean
class Solution {
    record Real(double lo, double hi) {}

    public double minmaxGasDist(int[] stations, int k) {
        int[] gap = IntStream.range(1, stations.length)
                .map(i -> stations[i] - stations[i - 1])
                .toArray();
        // New stations needed so that no gap exceeds d
        DoubleToLongFunction need = d -> Arrays.stream(gap)
                .mapToLong(g -> (long) Math.ceil(g / d) - 1)
                .sum();
        // A real-valued answer has no "empty range" to stop on,
        // so run a fixed 100 halvings: 10⁸ / 2¹⁰⁰ is far below
        // the 10⁻⁶ tolerance.
        double hi = Arrays.stream(gap).max().getAsInt();
        return Stream.iterate(new Real(0, hi), r -> {
                    double mid = (r.lo() + r.hi()) / 2;
                    return need.applyAsLong(mid) <= k
                            ? new Real(r.lo(), mid)
                            : new Real(mid, r.hi());
                })
                .skip(100)
                .findFirst().get().hi();
    }
}
The fold idiomJava 17 has no foldLeft. Stream.iterate(seed, step).skip(n).findFirst() works as one. It applies step exactly n times, so a record can index the array with its own counter and never read past the end. 1011 and 410 use it for the greedy packing, and the DP patterns later in this deck use it for rolling state.
The pattern, restated as codeSuppose the question asks for the smallest X that still works, and working only gets easier as X grows. Then the answer is firstTrue(lo, hi, x -> works(x)). Writing works as a pipeline makes you state it as a pure function of x. That purity is exactly the monotonicity argument the interviewer is waiting to hear.
