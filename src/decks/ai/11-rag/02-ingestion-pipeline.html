Ingestion is an ETL job: read, split, enrich, store, and make it repeatable

Spring AI models ingestion as DocumentReader → DocumentTransformers → DocumentWriter, and a VectorStore is a writer. The code is short; the engineering is in the details: stable ids so re-runs replace instead of duplicate, metadata for filters and citations, and removing chunks when the source changes or disappears.

KnowledgeIngestionService.java
package com.acme.ai.rag.ingest;

import java.nio.charset.StandardCharsets;
import java.time.Instant;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import org.springframework.ai.document.Document;
import org.springframework.ai.reader.tika.TikaDocumentReader;
import org.springframework.ai.transformer.splitter.TokenTextSplitter;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.ai.vectorstore.filter.FilterExpressionBuilder;
import org.springframework.core.io.Resource;
import org.springframework.stereotype.Service;

@Service
class KnowledgeIngestionService {

    record SourceDocument(String sourceId, String title, String url, Resource content,
                          String tenant, List<String> aclGroups, Instant lastModified) {}

    private final VectorStore vectorStore;
    private final TokenTextSplitter splitter = TokenTextSplitter.builder()
            .withChunkSize(500)            // tokens per chunk; the default is 800
            .withMinChunkSizeChars(200)    // look for a sentence end after this many characters
            .withMaxNumChunks(5_000)       // guard against runaway documents
            .build();

    KnowledgeIngestionService(VectorStore vectorStore) {
        this.vectorStore = vectorStore;
    }

    /** Idempotent: re-ingesting a source replaces its chunks. */
    int ingest(SourceDocument source) {
        List<Document> extracted = new TikaDocumentReader(source.content()).get();   // PDF, DOCX, HTML…
        List<Document> chunks = splitter.apply(extracted);

        List<Document> enriched = chunks.stream().map(chunk -> {
            Map<String, Object> md = new HashMap<>(chunk.getMetadata());   // keeps chunk_index
            md.put("sourceId", source.sourceId());
            md.put("title", source.title());
            md.put("url", source.url());
            md.put("tenant", source.tenant());
            md.put("aclGroup", source.aclGroups());       // a list: PgVector's JSONPath matches any element
            md.put("lastModified", source.lastModified().toString());
            String id = source.sourceId() + "#" + md.get("chunk_index");
            return Document.builder()
                    .id(UUID.nameUUIDFromBytes(id.getBytes(StandardCharsets.UTF_8)).toString())
                    .text(chunk.getText())
                    .metadata(md)
                    .build();
        }).toList();

        // the new version may have fewer chunks than the old one: remove the old ones first
        vectorStore.delete(new FilterExpressionBuilder().eq("sourceId", source.sourceId()).build());
        vectorStore.add(enriched);
        return enriched.size();
    }

    void remove(String sourceId) {
        vectorStore.delete(new FilterExpressionBuilder().eq("sourceId", sourceId).build());
    }
}
