Measure retrieval and generation separately, on a fixed set of real questions

"It looks good" is not a quality bar. Build a golden set of 50–200 real questions, each with the passage that answers it or a note that it is unanswerable. Run it on every change to prompts, chunking, models or thresholds. Score retrieval (did the right passage come back?) apart from generation (was the answer faithful and relevant?), because the fixes are different.

package com.acme.ai.rag;

import static org.assertj.core.api.Assertions.assertThat;

import java.util.List;
import org.junit.jupiter.api.Tag;
import org.junit.jupiter.api.Test;
import org.junit.jupiter.api.condition.EnabledIfEnvironmentVariable;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.evaluation.FactCheckingEvaluator;
import org.springframework.ai.chat.evaluation.RelevancyEvaluator;
import org.springframework.ai.document.Document;
import org.springframework.ai.evaluation.EvaluationRequest;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;

@Tag("eval")                                            // not part of the fast build
@SpringBootTest
@EnabledIfEnvironmentVariable(named = "OPENAI_API_KEY", matches = ".+")
class PolicyAssistantEvaluationTest {

    /** In practice: a JSON file of real questions taken from support tickets. */
    record Case(String question, String expectedSourceId, boolean answerable) {}

    static final List<Case> GOLDEN = List.of(
            new Case("Can I return a sofa after three weeks?", "returns-policy", true),
            new Case("How do I book a collection for a wardrobe?", "large-items", true),
            new Case("Do gift cards expire?", "gift-cards", true),
            new Case("What is your CEO's salary?", null, false));

    @Autowired VectorStore vectorStore;
    @Autowired CitedAnswerService assistant;
    @Autowired ChatClient.Builder judge;                  // ideally a different, stronger model

    @Test
    void retrievalFindsTheRightSource() {
        long hits = GOLDEN.stream().filter(Case::answerable).filter(c -> vectorStore
                .similaritySearch(SearchRequest.builder().query(c.question()).topK(5).build())
                .stream().anyMatch(d -> c.expectedSourceId().equals(d.getMetadata().get("sourceId"))))
                .count();
        double recallAt5 = (double) hits / GOLDEN.stream().filter(Case::answerable).count();
        assertThat(recallAt5).isGreaterThanOrEqualTo(0.9);
    }

    @Test
    void answersAreRelevantAndSupported() {
        var relevancy = RelevancyEvaluator.builder().chatClientBuilder(judge).build();
        var facts = FactCheckingEvaluator.builder(judge).build();
        int passed = 0, total = 0;
        for (Case c : GOLDEN) {
            var answer = assistant.ask(c.question());
            total++;
            if (!c.answerable()) {                         // must decline, with no sources
                if (answer.sources().isEmpty()) passed++;
                continue;
            }
            List<Document> context = vectorStore.similaritySearch(
                    SearchRequest.builder().query(c.question()).topK(5).build());
            var request = new EvaluationRequest(c.question(), context, answer.text());
            if (relevancy.evaluate(request).isPass() && facts.evaluate(request).isPass()) passed++;
        }
        assertThat((double) passed / total).isGreaterThanOrEqualTo(0.85);
    }
}
