One advisor turns a chat client into a question-answering client

QuestionAnswerAdvisor searches the vector store with the user's message, appends the matching chunks and a grounding instruction to it, and lets the call continue. It is the right first version: one dependency (spring-ai-vector-store-advisor), a search request you can tune, and a per-request filter for access control.

PolicyAssistant.java
package com.acme.ai.rag;

import java.util.List;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.client.ChatClientResponse;
import org.springframework.ai.chat.client.advisor.vectorstore.QuestionAnswerAdvisor;
import org.springframework.ai.document.Document;
import org.springframework.ai.vectorstore.SearchRequest;
import org.springframework.ai.vectorstore.VectorStore;
import org.springframework.stereotype.Service;

@Service
class PolicyAssistant {

    record Answer(String text, List<String> sources) {}

    private final ChatClient chatClient;

    PolicyAssistant(ChatClient.Builder builder, VectorStore vectorStore) {
        this.chatClient = builder
                .defaultSystem("You answer questions about Acme Retail's customer policies.")
                .defaultAdvisors(QuestionAnswerAdvisor.builder(vectorStore)
                        .searchRequest(SearchRequest.builder()
                                .topK(5)
                                .similarityThreshold(0.5)
                                .build())
                        .build())
                .build();
    }

    Answer ask(String question, String tenant) {
        ChatClientResponse response = chatClient.prompt()
                .user(question)
                .advisors(a -> a.param(QuestionAnswerAdvisor.FILTER_EXPRESSION,
                        "tenant == '" + tenant.replace("'", "") + "'"))
                .call()
                .chatClientResponse();

        @SuppressWarnings("unchecked")
        List<Document> used = (List<Document>) response.context()
                .getOrDefault(QuestionAnswerAdvisor.RETRIEVED_DOCUMENTS, List.of());

        return new Answer(
                response.chatResponse().getResult().getOutput().getText(),
                used.stream().map(d -> String.valueOf(d.getMetadata().get("title"))).distinct().toList());
    }
}
