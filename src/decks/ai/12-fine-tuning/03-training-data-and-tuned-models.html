Build the dataset from your systems; serve the tuned model like any other

The Java side of fine-tuning is mostly data engineering: select good examples from your systems, remove personal data, deduplicate, hold some back for evaluation, and write them in the provider's chat format (one JSON object per line). Once the tuning job finishes, a tuned model is just another model id in application.yml, so the application code does not change.

TrainingSetExporter.java
package com.acme.ai.finetune;

import java.io.BufferedWriter;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import java.util.regex.Pattern;
import java.util.stream.Stream;
import org.springframework.stereotype.Service;
import tools.jackson.databind.json.JsonMapper;

@Service
class TrainingSetExporter {

    record ResolvedTicket(String id, String customerMessage, String agentReply,
                          int satisfaction, boolean consentForTraining) {}

    interface TicketArchive {
        Stream<ResolvedTicket> resolvedSince(LocalDate date);
    }

    /** One line of the chat fine-tuning format: {"messages":[{"role":…,"content":…},…]} */
    record Message(String role, String content) {}

    record Example(List<Message> messages) {}

    record Counts(int train, int holdout, int skipped) {}

    private static final String SYSTEM = "You are Acme Retail's support assistant.";
    private static final Pattern EMAIL = Pattern.compile("[\\w.+-]+@[\\w-]+(\\.[\\w-]+)+");
    private static final Pattern NUMBER = Pattern.compile("\\b\\d[\\d -]{6,}\\d\\b");

    private final TicketArchive archive;
    private final JsonMapper json;

    TrainingSetExporter(TicketArchive archive, JsonMapper json) {
        this.archive = archive;
        this.json = json;
    }

    Counts export(LocalDate since, Path dir) throws IOException {
        Set<String> seen = new HashSet<>();
        int train = 0, holdout = 0, skipped = 0;
        try (Stream<ResolvedTicket> tickets = archive.resolvedSince(since);
             BufferedWriter trainOut = Files.newBufferedWriter(dir.resolve("train.jsonl"));
             BufferedWriter holdoutOut = Files.newBufferedWriter(dir.resolve("holdout.jsonl"))) {
            for (ResolvedTicket t : (Iterable<ResolvedTicket>) tickets::iterator) {
                String question = redact(t.customerMessage());
                boolean good = t.consentForTraining() && t.satisfaction() >= 4;
                if (!good || !seen.add(question.toLowerCase().strip())) {
                    skipped++;                      // unconsented, poorly rated or duplicate
                    continue;
                }
                String line = json.writeValueAsString(new Example(List.of(
                        new Message("system", SYSTEM),
                        new Message("user", question),
                        new Message("assistant", redact(t.agentReply())))));
                boolean isHoldout = Math.floorMod(t.id().hashCode(), 10) == 0;   // stable 10%
                BufferedWriter out = isHoldout ? holdoutOut : trainOut;
                out.write(line);
                out.newLine();
                if (isHoldout) holdout++; else train++;
            }
        }
        return new Counts(train, holdout, skipped);
    }

    static String redact(String text) {
        return NUMBER.matcher(EMAIL.matcher(text).replaceAll("[EMAIL]")).replaceAll("[NUMBER]");
    }
}
