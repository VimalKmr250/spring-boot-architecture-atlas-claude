Cost control: fewer tokens, cheaper tokens, cached tokens, and a budget per user

LLM cost is tokens times price, and both are under your control. Most savings come from sending less (trimmed history, fewer and better chunks), routing easy work to small models, and reusing work through caching. A per-user or per-tenant budget turns a runaway loop or an abusive client from an invoice surprise into an error message.

TokenBudgetAdvisor.java
package com.acme.ai.ops;

import java.time.LocalDate;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.atomic.AtomicLong;
import org.springframework.ai.chat.client.ChatClientRequest;
import org.springframework.ai.chat.client.ChatClientResponse;
import org.springframework.ai.chat.client.advisor.api.AdvisorChain;
import org.springframework.ai.chat.client.advisor.api.BaseAdvisor;
import org.springframework.ai.chat.metadata.Usage;
import org.springframework.core.Ordered;

/** Daily token budget per user. In production keep the counters in Redis, not in memory. */
public class TokenBudgetAdvisor implements BaseAdvisor {

    public static final String USER = "budget.user";

    public static class BudgetExceededException extends RuntimeException {
        private static final long serialVersionUID = 1L;

        BudgetExceededException(String user) {
            super("Daily AI budget used up for " + user);
        }
    }

    private final long dailyTokens;
    private final Map<String, AtomicLong> spent = new ConcurrentHashMap<>();

    public TokenBudgetAdvisor(long dailyTokens) {
        this.dailyTokens = dailyTokens;
    }

    @Override
    public ChatClientRequest before(ChatClientRequest request, AdvisorChain chain) {
        String user = (String) request.context().get(USER);
        if (user != null && counter(user).get() >= dailyTokens) {
            throw new BudgetExceededException(user);       // no model call is made
        }
        return request;
    }

    @Override
    public ChatClientResponse after(ChatClientResponse response, AdvisorChain chain) {
        String user = (String) response.context().get(USER);
        if (user != null && response.chatResponse() != null) {
            Usage usage = response.chatResponse().getMetadata().getUsage();
            Integer total = usage == null ? null : usage.getTotalTokens();
            if (total != null) {
                counter(user).addAndGet(total);
            }
        }
        return response;
    }

    @Override
    public int getOrder() {
        return Ordered.HIGHEST_PRECEDENCE + 50;            // first in, last out
    }

    private AtomicLong counter(String user) {
        return spent.computeIfAbsent(user + "|" + LocalDate.now(), k -> new AtomicLong());
    }
}
Usage.advisors(new TokenBudgetAdvisor(200_000)) on the client, and .advisors(a -> a.param(TokenBudgetAdvisor.USER, userId)) per request. Map BudgetExceededException to HTTP 429 with a friendly message. It sits outside the tool loop, and that is fine: verified on Spring AI 2.0.1, the final response's usage is the sum of every model call in the loop (a tool call plus the answer at 110 tokens each reported 220). With a 300-token budget, the third request was refused before any model call.
