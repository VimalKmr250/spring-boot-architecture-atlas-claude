Observe tokens, latency, errors and quality, per feature, with content kept out of logs

Spring AI instruments ChatClient, advisors, models, tools and vector stores with Micrometer, so metrics and traces appear as soon as Actuator and a tracing bridge are on the classpath. Two things are yours to add: a business dimension (which feature spent the tokens?) and quality signals (did the answer help?). Prompt and completion content stays out of telemetry by default; keep it that way unless a reviewed decision says otherwise.

application.yml
management:
  endpoints:
    web:
      exposure:
        include: health, metrics, prometheus
  tracing:
    sampling:
      probability: 0.1          # 10% of requests traced end to end
spring:
  ai:
    chat:
      observations:
        log-prompt: false       # the defaults: content stays out of
        log-completion: false   # logs and spans
      client:
        observations:
          log-prompt: false
    tools:
      observations:
        include-content: false  # tool arguments and results

AiUsageMetrics.java · tokens per business feature
package com.acme.ai.ops;

import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.MeterRegistry;
import org.springframework.ai.chat.metadata.Usage;
import org.springframework.ai.chat.model.ChatResponse;
import org.springframework.stereotype.Component;

@Component
public class AiUsageMetrics {

    private final MeterRegistry registry;

    AiUsageMetrics(MeterRegistry registry) {
        this.registry = registry;
    }

    /** Call with every ChatResponse; feature is low-cardinality, e.g. "support-bot". */
    public void record(String feature, ChatResponse response) {
        Usage usage = response.getMetadata().getUsage();
        String model = response.getMetadata().getModel();
        count(feature, model, "input", usage.getPromptTokens());
        count(feature, model, "output", usage.getCompletionTokens());
    }

    private void count(String feature, String model, String type, Integer tokens) {
        if (tokens == null) {
            return;
        }
        Counter.builder("acme.ai.tokens")
                .tag("feature", feature)
                .tag("model", model == null ? "unknown" : model)
                .tag("type", type)
                .register(registry)
                .increment(tokens);
    }
}
