Annotate a method with @Tool; pass request data through ToolContext

Any bean method can be a tool. Spring AI builds the JSON schema from the parameters, serialises the return value to JSON for the model, and runs the method when the model asks. A ToolContext parameter is excluded from the schema: it carries data from your code to the tool, such as the authenticated customer, that the model must never see or choose.

OrderTools.java
package com.acme.ai.tools;

import java.time.LocalDate;
import java.util.List;
import java.util.Optional;
import org.springframework.ai.chat.model.ToolContext;
import org.springframework.ai.tool.annotation.Tool;
import org.springframework.ai.tool.annotation.ToolParam;
import org.springframework.stereotype.Component;

@Component
class OrderTools {

    record OrderStatus(String orderId, String status, String carrier, LocalDate estimatedDelivery) {}

    /** Your existing order service, behind a port. */
    interface OrderLookup {
        Optional<OrderStatus> findForCustomer(String orderId, String customerId);
        List<OrderStatus> recentForCustomer(String customerId, int limit);
    }

    private final OrderLookup orders;

    OrderTools(OrderLookup orders) {
        this.orders = orders;
    }

    @Tool(description = """
            Returns status, carrier and estimated delivery date of one of the
            current customer's orders. Use it for any question about a specific order.""")
    OrderStatus getOrderStatus(@ToolParam(description = "Order id, for example 1042") String orderId,
                               ToolContext context) {
        return orders.findForCustomer(orderId, customerId(context))
                .orElseThrow(() -> new IllegalArgumentException(
                        "No order " + orderId + " for this customer"));
    }

    @Tool(description = "Lists the current customer's most recent orders, newest first.")
    List<OrderStatus> recentOrders(@ToolParam(description = "How many orders, 1 to 10") int limit,
                                   ToolContext context) {
        return orders.recentForCustomer(customerId(context), Math.clamp(limit, 1, 10));
    }

    private static String customerId(ToolContext context) {
        return (String) context.getContext().get("customerId");
    }
}
OrderAssistant.java
package com.acme.ai.tools;

import java.util.Map;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.stereotype.Service;

@Service
class OrderAssistant {

    private final ChatClient chatClient;

    OrderAssistant(ChatClient.Builder builder, OrderTools orderTools) {
        this.chatClient = builder
                .defaultSystem("""
                        You are Acme Retail's order assistant. Use the tools for any
                        question about orders; never guess order details.""")
                .defaultTools(orderTools)
                .build();
    }

    String ask(String question, String customerId) {
        return chatClient.prompt()
                .user(question)
                .toolContext(Map.of("customerId", customerId))   // from the session, not the model
                .call()
                .content();
    }
}

  },
  "required" : [ "orderId" ],
  "additionalProperties" : false
}
Notice what is missingNo customerId and no ToolContext in the schema. The model can only choose the order id; whose orders it may see is fixed by your code. That one design choice closes the most common tool vulnerability: a user persuading the model to look up somebody else's data.
