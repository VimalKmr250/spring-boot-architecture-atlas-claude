Put conversations in your database, size the window, and expire them

Add the JDBC chat memory starter and the auto-configured ChatMemory writes to a SPRING_AI_CHAT_MEMORY table in your existing database. Redis, Cassandra, MongoDB and Neo4j starters work the same way. Two things are still yours: the window size, and how long conversations live, because chat transcripts are personal data.

The PostgreSQL schema shipped in Spring AI 2.0.1
CREATE TABLE IF NOT EXISTS SPRING_AI_CHAT_MEMORY (
    conversation_id VARCHAR(36) NOT NULL,
    content TEXT NOT NULL,
    type VARCHAR(10) NOT NULL
        CHECK (type IN ('USER', 'ASSISTANT', 'SYSTEM', 'TOOL')),
    "timestamp" TIMESTAMP NOT NULL,
    sequence_id BIGINT NOT NULL
);
CREATE INDEX IF NOT EXISTS SPRING_AI_CHAT_MEMORY_CONVERSATION_ID_TIMESTAMP_IDX
    ON SPRING_AI_CHAT_MEMORY(conversation_id, "timestamp");
CREATE INDEX IF NOT EXISTS SPRING_AI_CHAT_MEMORY_CONVERSATION_ID_SEQUENCE_ID_IDX
    ON SPRING_AI_CHAT_MEMORY(conversation_id, sequence_id);
ChatMemoryConfig.java
package com.acme.ai.memory.jdbc;

import java.time.Duration;
import java.time.LocalDateTime;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.ai.chat.memory.ChatMemory;
import org.springframework.ai.chat.memory.ChatMemoryRepository;
import org.springframework.ai.chat.memory.MessageWindowChatMemory;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.EnableScheduling;
import org.springframework.scheduling.annotation.Scheduled;

@Configuration
@EnableScheduling
class ChatMemoryConfig {

    private static final Logger log = LoggerFactory.getLogger(ChatMemoryConfig.class);
    private static final Duration RETENTION = Duration.ofDays(30);

    private final JdbcTemplate jdbc;

    ChatMemoryConfig(JdbcTemplate jdbc) {
        this.jdbc = jdbc;
    }

    /** Replaces the auto-configured 20-message window; the JDBC repository is still injected. */
    @Bean
    ChatMemory chatMemory(ChatMemoryRepository repository) {
        return MessageWindowChatMemory.builder()
                .chatMemoryRepository(repository)
                .maxMessages(12)
                .build();
    }

    /** Conversations are personal data: keep them only as long as the policy says. */
    @Scheduled(cron = "0 15 3 * * *")
    void expireOldConversations() {
        int deleted = jdbc.update("""
                DELETE FROM SPRING_AI_CHAT_MEMORY
                WHERE conversation_id IN (
                    SELECT conversation_id FROM SPRING_AI_CHAT_MEMORY
                    GROUP BY conversation_id
                    HAVING MAX("timestamp") < ?)""",
                LocalDateTime.now().minus(RETENTION));
        log.info("expired {} chat memory rows", deleted);
    }
}
1.x migrating an existing tableSpring AI 2.0 added sequence_id and an index on (conversation_id, sequence_id) to order messages reliably. A table created by 1.x needs both before you upgrade: write a Flyway migration that adds the column, backfills it from "timestamp" order, and creates the index.
Window size is a trade-offLarger windows remember more and cost more on every turn. 10–20 messages covers most support chats; beyond that, summarise older turns (next slide).
