A multi-turn chat API with memory, ownership checks and a delete

The auto-configured ChatMemory bean keeps a 20-message window per conversation, stored in memory unless a repository starter is present. Add the advisor as a default, pass the conversation id on each request, and treat the id like any other resource identifier: issued by the server, owned by a user, checked on every access, deletable on request.

SupportChatController.java
package com.acme.ai.memory;

import jakarta.validation.Valid;
import jakarta.validation.constraints.NotBlank;
import jakarta.validation.constraints.Size;
import java.security.Principal;
import java.util.Map;
import java.util.UUID;
import java.util.concurrent.ConcurrentHashMap;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.ai.chat.client.advisor.MessageChatMemoryAdvisor;
import org.springframework.ai.chat.memory.ChatMemory;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Component;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.server.ResponseStatusException;

@RestController
@RequestMapping("/api/conversations")
class SupportChatController {

    record UserText(@NotBlank @Size(max = 2_000) String text) {}

    record Reply(String conversationId, String text) {}

    private final ChatClient chatClient;
    private final ChatMemory chatMemory;
    private final ConversationOwners owners;

    SupportChatController(ChatClient.Builder builder, ChatMemory chatMemory,
                          ConversationOwners owners) {
        this.chatClient = builder
                .defaultSystem("You are Acme Retail's support assistant. Be brief.")
                .defaultAdvisors(MessageChatMemoryAdvisor.builder(chatMemory).build())
                .build();
        this.chatMemory = chatMemory;
        this.owners = owners;
    }

    @PostMapping
    Reply start(@Valid @RequestBody UserText first, Principal user) {
        return send(owners.open(user.getName()), first, user);
    }

    @PostMapping("/{id}/messages")
    Reply send(@PathVariable String id, @Valid @RequestBody UserText message, Principal user) {
        owners.requireOwner(id, user.getName());
        String text = chatClient.prompt()
                .user(message.text())
                .advisors(a -> a.param(ChatMemory.CONVERSATION_ID, id))
                .call()
                .content();
        return new Reply(id, text);
    }

    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    void delete(@PathVariable String id, Principal user) {
        owners.requireOwner(id, user.getName());
        chatMemory.clear(id);                       // the user's "forget this" button
    }
}

@Component
class ConversationOwners {

    private final Map<String, String> ownerById = new ConcurrentHashMap<>();   // a table in production

    String open(String userName) {
        String id = UUID.randomUUID().toString();   // 36 chars: fits the JDBC schema
        ownerById.put(id, userName);
        return id;
    }

    void requireOwner(String id, String userName) {
        if (!userName.equals(ownerById.get(id))) {
            throw new ResponseStatusException(HttpStatus.NOT_FOUND);   // do not reveal it exists
        }
    }
}

The window counts messages, not tokensA 20-message window of short questions is a few hundred tokens; 20 messages that each paste a contract can overflow the context window. Size the window from real conversations, and cap the size of each user message at the API, as @Size does here.
Memory and toolsIn 2.0 the memory advisor sits outside the tool loop, so one user turn is stored as one user message and one final answer, not the intermediate tool calls. If you need the full trace, including tool calls, for audit, look at the community spring-ai-session project, an event-sourced replacement for ChatMemory.
