The weekly report job: snapshot, draft, number check, save for approval

The job is a plain scheduled Spring component. JdbcClient queries build a typed snapshot with values already rounded for display, the model turns it into a structured draft, and a checker rejects any draft containing a figure that is not in the snapshot. Approved drafts are published by a separate step that only a person can trigger.

WeeklyReportJob.java
package com.acme.ai.usecases.reports;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.DayOfWeek;
import java.time.LocalDate;
import java.time.temporal.TemporalAdjusters;
import java.util.List;
import java.util.Set;
import java.util.regex.MatchResult;
import java.util.regex.Pattern;
import java.util.stream.Collectors;
import org.springframework.ai.chat.client.ChatClient;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import tools.jackson.databind.json.JsonMapper;

@Component
class WeeklyReportJob {

    record CategorySales(String category, BigDecimal revenueK, BigDecimal changePct) {}

    record MetricsSnapshot(BigDecimal revenueK, BigDecimal revenueChangePct, long orders,
                           BigDecimal returnRatePct, List<CategorySales> topCategories) {}

    record Section(String heading, String body) {}

    record ReportDraft(String headline, List<Section> sections, List<String> caveats) {}

    interface Drafts {
        void saveForApproval(LocalDate weekStart, MetricsSnapshot data, ReportDraft draft);
    }

    private static final Pattern NUMBER = Pattern.compile("\\d[\\d,]*(?:\\.\\d+)?");

    private final JdbcClient jdbc;
    private final ChatClient chatClient;
    private final JsonMapper json;
    private final Drafts drafts;

    WeeklyReportJob(JdbcClient jdbc, ChatClient.Builder builder, JsonMapper json, Drafts drafts) {
        this.jdbc = jdbc;
        this.json = json;
        this.drafts = drafts;
        this.chatClient = builder
                .defaultSystem("""
                        You write Acme Retail's weekly trading commentary for store managers.
                        Use ONLY numbers that appear in the data, copied exactly; never
                        calculate, round, convert or estimate. Do not mention dates.
                        Explain the biggest movements first. Add caveats for anything unusual.""")
                .build();
    }

    @Scheduled(cron = "0 0 6 * * MON", zone = "Europe/London")
    void run() {
        LocalDate weekStart = LocalDate.now().with(TemporalAdjusters.previous(DayOfWeek.MONDAY));
        MetricsSnapshot data = snapshot(weekStart);
        ReportDraft draft = draft(data);
        if (!unknownNumbers(draft, data).isEmpty()) {
            draft = draft(data);                                    // one retry
            List<String> unknown = unknownNumbers(draft, data);
            if (!unknown.isEmpty()) {
                throw new IllegalStateException("Draft has numbers not in the data: " + unknown);
            }
        }
        drafts.saveForApproval(weekStart, data, draft);             // status DRAFT
    }

    MetricsSnapshot snapshot(LocalDate weekStart) {
        BigDecimal revenue = jdbc.sql("""
                        SELECT COALESCE(SUM(total), 0) FROM orders
                        WHERE placed_at >= :from AND placed_at < :to""")
                .param("from", weekStart).param("to", weekStart.plusWeeks(1))
                .query(BigDecimal.class).single();
        // … the same pattern for the previous week, orders, returns and categories
        return new MetricsSnapshot(thousands(revenue), BigDecimal.ZERO, 0, BigDecimal.ZERO, List.of());
    }

    ReportDraft draft(MetricsSnapshot data) {
        return chatClient.prompt()
                .user(u -> u.text("Data (the only source of numbers):\n{data}")
                        .param("data", json.writeValueAsString(data)))
                .call()
                .entity(ReportDraft.class);
    }

    /** Every number in the draft must appear in the data; small counts ("top 3") are allowed. */
    List<String> unknownNumbers(ReportDraft draft, MetricsSnapshot data) {
        Set<String> allowed = numbers(json.writeValueAsString(data)).collect(Collectors.toSet());
        String text = draft.headline() + " "
                + draft.sections().stream().map(s -> s.heading() + " " + s.body())
                        .collect(Collectors.joining(" "))
                + " " + String.join(" ", draft.caveats());
        return numbers(text)
                .filter(n -> !allowed.contains(n))
                .filter(n -> new BigDecimal(n).compareTo(BigDecimal.TEN) > 0)
                .distinct()
                .toList();
    }

    private static java.util.stream.Stream<String> numbers(String text) {
        return NUMBER.matcher(text).results().map(MatchResult::group)
                .map(n -> new BigDecimal(n.replace(",", "")).stripTrailingZeros().toPlainString());
    }

    private static BigDecimal thousands(BigDecimal value) {
        return value.divide(BigDecimal.valueOf(1000), 1, RoundingMode.HALF_UP);
    }
}

The check is a net, not a proofIt catches invented and recalculated figures, the most damaging error, but the small-count allowance lets "about 9% above plan" through, and it cannot tell whether "rose" should have been "fell". Tighten it where it matters (for example, check every number followed by %), and keep the human approval step. Track how often drafts are rejected or heavily edited; that is the quality metric.
Pre-round in codeModels round and convert units when they think it reads better. Give them display-ready values (412.6 in thousands, 8.3 per cent) and forbid conversions, so the exact-match check stays meaningful.
