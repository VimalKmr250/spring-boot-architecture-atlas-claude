Splitwise: netting debts per pair, simplifying them greedily, and a command-line run

Recording "bo owes ana 300" first cancels against any "ana owes bo" that already exists, so each pair of people has at most one debt. The simplifier ignores pairs entirely. It works from each person's net balance and repeatedly lets the biggest debtor pay the biggest creditor. Each payment settles at least one person, so n people never need more than n − 1 payments.

final class Ledger {
    private final Map<String, Map<String, Long>> owes = new TreeMap<>();  // debtor -> creditor
    private final List<Expense> expenses = new ArrayList<>();

    synchronized Expense add(String payer, long total, SplitRule rule) {
        if (total <= 0) throw new IllegalArgumentException("total must be positive");
        Map<String, Long> shares = rule.shares(total);           // validates first
        var e = new Expense("E" + (expenses.size() + 1), payer, total, shares);
        shares.forEach((person, share) -> {
            if (!person.equals(payer)) owe(person, payer, share);
        });
        expenses.add(e);
        return e;
    }

    synchronized void settle(String from, String to, long amount) {
        if (amount <= 0) throw new IllegalArgumentException("amount must be positive");
        owe(to, from, amount);                     // a payment is a debt reversed
    }

    private void owe(String debtor, String creditor, long amount) {
        long reverse = owes.getOrDefault(creditor, Map.of()).getOrDefault(debtor, 0L);
        long offset = Math.min(reverse, amount);   // cancel the opposite debt first
        adjust(creditor, debtor, -offset);
        adjust(debtor, creditor, amount - offset);
    }

    private void adjust(String debtor, String creditor, long delta) {
        if (delta == 0) return;
        Map<String, Long> row = owes.computeIfAbsent(debtor, k -> new TreeMap<>());
        if (row.merge(creditor, delta, Long::sum) == 0) row.remove(creditor);
        if (row.isEmpty()) owes.remove(debtor);
    }

    synchronized List<Transfer> debts() {
        List<Transfer> out = new ArrayList<>();
        owes.forEach((d, row) -> row.forEach((c, amt) -> out.add(new Transfer(d, c, amt))));
        return out;
    }

    synchronized Map<String, Long> netBalances() {                 // + is owed
        Map<String, Long> net = new TreeMap<>();
        owes.forEach((d, row) -> row.forEach((c, amt) -> {
            net.merge(c, amt, Long::sum);
            net.merge(d, -amt, Long::sum);
        }));
        return net;
    }
}

final class DebtSimplifier {
    private record Party(String who, long amount) {}

    static List<Transfer> simplify(Map<String, Long> net) {
        Comparator<Party> biggestFirst = Comparator.comparingLong(Party::amount)
                .reversed().thenComparing(Party::who);             // ties: by name
        var creditors = new PriorityQueue<>(biggestFirst);
        var debtors = new PriorityQueue<>(biggestFirst);
        net.forEach((who, amt) -> {
            if (amt > 0) creditors.add(new Party(who, amt));
            if (amt < 0) debtors.add(new Party(who, -amt));
        });
        List<Transfer> out = new ArrayList<>();
        while (!creditors.isEmpty() && !debtors.isEmpty()) {
            Party c = creditors.poll(), d = debtors.poll();
            long amt = Math.min(c.amount(), d.amount());
            out.add(new Transfer(d.who(), c.who(), amt));           // one settles fully
            if (c.amount() > amt) creditors.add(new Party(c.who(), c.amount() - amt));
            if (d.amount() > amt) debtors.add(new Party(d.who(), d.amount() - amt));
        }
        return out;
    }
}

public class SplitwiseDemo {
    static final Ledger ledger = new Ledger();

    static void run(String line) {
        System.out.println("> " + line);
        String[] t = line.trim().split("\\s+");
        try {
            switch (t[0]) {
                case "EXPENSE" -> {                 // EXPENSE payer total TYPE args
                    List<String> people = new ArrayList<>();
                    List<Long> values = new ArrayList<>();
                    for (int i = 4; i < t.length; i++) {
                        String[] kv = t[i].split("=");
                        people.add(kv[0]);
                        if (kv.length == 2)
                            values.add(t[3].equals("EXACT") ? Money.parse(kv[1]) : Long.parseLong(kv[1]));
                    }
                    var e = ledger.add(t[1], Money.parse(t[2]), SplitRules.of(t[3], people, values));
                    var parts = new StringJoiner(", ");
                    e.shares().forEach((p, s) -> parts.add(p + " " + Money.format(s)));
                    System.out.println("  " + e.id() + " shares: " + parts);
                }
                case "SETTLE" -> ledger.settle(t[1], t[2], Money.parse(t[3]));
                case "SHOW" -> ledger.debts().forEach(d -> System.out.println(
                        "  " + d.from() + " owes " + d.to() + " " + Money.format(d.amount())));
                case "SIMPLIFY" -> DebtSimplifier.simplify(ledger.netBalances())
                        .forEach(x -> System.out.println("  " + x));
                default -> throw new IllegalArgumentException("unknown command " + t[0]);
            }
        } catch (IllegalArgumentException | ArithmeticException e) {
            System.out.println("  error: " + e.getMessage());
        }
    }

    public static void main(String[] args) {
        """
        EXPENSE ana 1200.00 EQUAL ana bo cy
        EXPENSE bo 1000.00 EXACT ana=300.00 bo=200.00 cy=500.00
        EXPENSE cy 999.99 PERCENT ana=50 dee=50
        EXPENSE dee 100.00 EQUAL ana bo dee
        EXPENSE ana 600.00 SHARES bo=1 cy=2
        EXPENSE ana 500.00 EXACT bo=100.00 cy=100.00
        EXPENSE ana 500.00 PERCENT bo=60 cy=30
        SHOW
        SIMPLIFY
        SETTLE cy ana 300.00
        SHOW
        """.lines().forEach(SplitwiseDemo::run);
    }
}
