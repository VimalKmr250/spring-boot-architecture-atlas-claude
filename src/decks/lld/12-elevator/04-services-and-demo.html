Elevator: two dispatch strategies, the controller, and a run with riders

The nearest-car dispatcher costs each car: the plain distance if it is idle or the call is on its way in the same direction, or the length of its detour (to the end of its run and back) if not. The zoned dispatcher ignores position entirely. The controller swaps them at runtime without touching a car.

interface Dispatcher {
    ElevatorCar choose(List<ElevatorCar> cars, HallCall call);
}

final class NearestCarDispatcher implements Dispatcher {
    public ElevatorCar choose(List<ElevatorCar> cars, HallCall call) {
        return cars.stream()
                .min(Comparator.comparingInt((ElevatorCar c) -> cost(c, call))
                        .thenComparing(ElevatorCar::id))          // ties: stable
                .orElseThrow();
    }

    static int cost(ElevatorCar c, HallCall call) {
        int gap = call.floor() - c.floor();
        return switch (c.direction()) {
            case IDLE -> Math.abs(gap);
            case UP   -> gap >= 0 && call.direction() == Direction.UP ? gap : detour(c, call);
            case DOWN -> gap <= 0 && call.direction() == Direction.DOWN ? -gap : detour(c, call);
        };
    }

    private static int detour(ElevatorCar c, HallCall call) {
        int turn = c.furthestStop();                 // finish the run, come back
        return Math.abs(turn - c.floor()) + Math.abs(turn - call.floor());
    }
}

record ZonedDispatcher(int floorsPerCar) implements Dispatcher {
    public ElevatorCar choose(List<ElevatorCar> cars, HallCall call) {
        return cars.get(Math.min(call.floor() / floorsPerCar, cars.size() - 1));
    }
}

final class ElevatorSystem {
    private final int floors;
    private final List<ElevatorCar> cars;
    private final List<ArrivalListener> listeners = new ArrayList<>();
    private Dispatcher dispatcher;
    private int tick;

    ElevatorSystem(int floors, List<ElevatorCar> cars, Dispatcher dispatcher) {
        this.floors = floors;
        this.cars = List.copyOf(cars);
        this.dispatcher = dispatcher;
    }

    synchronized void subscribe(ArrivalListener l) { listeners.add(l); }
    synchronized void useDispatcher(Dispatcher d) { dispatcher = Objects.requireNonNull(d); }

    synchronized String hallCall(int floor, Direction direction) {
        checkFloor(floor);
        ElevatorCar car = dispatcher.choose(cars, new HallCall(floor, direction));
        car.addStop(floor);
        return car.id();
    }

    synchronized void carCall(String carId, int floor) {
        checkFloor(floor);
        cars.stream().filter(c -> c.id().equals(carId)).findFirst()
            .orElseThrow(() -> new IllegalArgumentException("no car " + carId))
            .addStop(floor);
    }

    synchronized void tick() {
        tick++;
        for (ElevatorCar c : cars)
            c.step(tick, (t, car, f) -> listeners.forEach(l -> l.stopped(t, car, f)));
    }

    synchronized boolean idle() { return cars.stream().noneMatch(ElevatorCar::hasStops); }
    synchronized List<ElevatorCar> cars() { return cars; }

    private void checkFloor(int floor) {
        if (floor < 0 || floor >= floors) throw new IllegalArgumentException("no floor " + floor);
    }
}
ElevatorDemo.java · 3 of 3 · the demo
public class ElevatorDemo {
    public static void main(String[] args) {
        var sys = new ElevatorSystem(10,
                List.of(new ElevatorCar("A", 0), new ElevatorCar("B", 9)),
                new NearestCarDispatcher());
        Map<String, Integer> riders = new HashMap<>(Map.of("A@3", 8, "B@4", 0));
        sys.subscribe((t, car, floor) -> {
            System.out.printf("t%-2d %s stops at %d%n", t, car, floor);
            Integer to = riders.remove(car + "@" + floor);    // a rider gets in
            if (to != null) {
                System.out.println("    rider in " + car + " presses " + to);
                sys.carCall(car, to);
            }
        });

        sys.carCall("A", 6);
        sys.tick();
        sys.tick();
        System.out.println("now " + sys.cars());
        System.out.println("hall 4 DOWN -> " + sys.hallCall(4, Direction.DOWN));
        System.out.println("hall 3 UP   -> " + sys.hallCall(3, Direction.UP));
        while (!sys.idle()) sys.tick();
        System.out.println("now " + sys.cars());

        System.out.println("nearest: hall 7 DOWN -> " + sys.hallCall(7, Direction.DOWN));
        sys.useDispatcher(new ZonedDispatcher(5));
        System.out.println("zoned:   hall 1 UP   -> " + sys.hallCall(1, Direction.UP));
        while (!sys.idle()) sys.tick();
        System.out.println("now " + sys.cars());
    }
}
