Four patterns you should recognise, and write when asked

These come up less often as the centre of a design, but interviewers use them as follow-ups. "How would you let callers walk your collection?" is iterator. "The flights all talk to each other" is mediator. "Add undo without exposing the state" is memento. "Add operations to this closed hierarchy" is visitor, which Java 21's pattern-matching switch mostly replaces.

FourPatterns.java · 1 of 2 · iterator and mediator
import java.util.*;

record Range(int from, int to, int step) implements Iterable<Integer> {
    public Iterator<Integer> iterator() {          // ITERATOR: callers walk
        return new Iterator<>() {                  // it, never see its insides
            private int next = from;
            public boolean hasNext() { return next < to; }
            public Integer next() {
                if (!hasNext()) throw new NoSuchElementException();
                int v = next;
                next += step;
                return v;
            }
        };
    }
}

final class Tower {                                // MEDIATOR: flights talk
    private final List<Flight> flights = new ArrayList<>();   // to the tower,
    private Flight onRunway;                                  // not each other
    void register(Flight f) { flights.add(f); }
    void requestLanding(Flight f) {
        if (onRunway == null) { onRunway = f; f.hear("cleared to land"); }
        else f.hear("hold, runway busy with " + onRunway.id());
    }
    void landed(Flight f) {
        onRunway = null;
        for (Flight o : flights) if (o != f) o.hear(f.id() + " has vacated the runway");
    }
}

record Flight(String id, Tower tower) {
    void land()   { tower.requestLanding(this); }
    void landed() { tower.landed(this); }
    void hear(String msg) { System.out.println("  " + id + " <- " + msg); }
}
FourPatterns.java · 2 of 2 · memento and visitor
final class TextBox {                              // MEMENTO: the originator
    private String text = "";
    private int cursor;
    void type(String s) {
        text = text.substring(0, cursor) + s + text.substring(cursor);
        cursor += s.length();
    }
    Snapshot save() { return new Snapshot(text, cursor); }
    void restore(Snapshot s) { text = s.text; cursor = s.cursor; }
    @Override public String toString() { return text + " (cursor " + cursor + ")"; }

    static final class Snapshot {                  // opaque to the caretaker:
        private final String text;                 // no public accessors
        private final int cursor;
        private Snapshot(String text, int cursor) { this.text = text; this.cursor = cursor; }
    }
}

sealed interface Expr permits Num, Add, Mul {
    <R> R accept(ExprVisitor<R> v);                // VISITOR: double dispatch
}
record Num(int v) implements Expr { public <R> R accept(ExprVisitor<R> x) { return x.num(this); } }
record Add(Expr l, Expr r) implements Expr { public <R> R accept(ExprVisitor<R> x) { return x.add(this); } }
record Mul(Expr l, Expr r) implements Expr { public <R> R accept(ExprVisitor<R> x) { return x.mul(this); } }

interface ExprVisitor<R> { R num(Num n); R add(Add a); R mul(Mul m); }

final class Show implements ExprVisitor<String> {  // one operation per class
    public String num(Num n) { return String.valueOf(n.v()); }
    public String add(Add a) { return "(" + a.l().accept(this) + " + " + a.r().accept(this) + ")"; }
    public String mul(Mul m) { return a(m.l()) + " * " + a(m.r()); }
    private String a(Expr e) { return e.accept(this); }
}

public class FourPatterns {
    static int eval(Expr e) {                      // the Java 21 alternative:
        return switch (e) {                        // a sealed switch with
            case Num(int v) -> v;                  // record patterns
            case Add(Expr l, Expr r) -> eval(l) + eval(r);
            case Mul(Expr l, Expr r) -> eval(l) * eval(r);
        };
    }

    public static void main(String[] args) {
        List<Integer> seen = new ArrayList<>();
        for (int i : new Range(0, 10, 3)) seen.add(i);
        System.out.println("iterator: " + seen);

        System.out.println("mediator:");
        var tower = new Tower();
        var a = new Flight("AI101", tower);
        var b = new Flight("6E202", tower);
        tower.register(a); tower.register(b);
        a.land(); b.land(); a.landed(); b.land();

        var box = new TextBox();
        box.type("hello");
        var saved = box.save();
        box.type(" world");
        System.out.print("memento: " + box);
        box.restore(saved);
        System.out.println(" -> " + box);

        Expr e = new Mul(new Add(new Num(2), new Num(3)), new Num(4));
        System.out.println("visitor: " + e.accept(new Show()) + " = " + eval(e));
    }
}

Visitor in modern JavaThe visitor exists because older Java could not switch on type safely. With a sealed hierarchy, eval above does the same job in five lines, and the compiler still rejects a switch that misses a new subtype. Mention the visitor, then say you would use the sealed switch since 21. Keep the visitor for code that must compile on Java 17 or earlier.
