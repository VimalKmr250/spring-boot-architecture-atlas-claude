Parking lot: the pricing strategies, the lot service, and a run through every rule

The lot tries floors in order, records the ticket, and on exit removes the ticket before anything else, so a ticket can be redeemed only once even if two exit gates scan it together. Time comes from an injected Clock, so the demo can move time forward and print the same fees on every run.

interface PricingStrategy {
    long fee(VehicleType type, Duration stayed);
}

final class HourlyPricing implements PricingStrategy {
    private final Map<VehicleType, Long> perHour;
    HourlyPricing(Map<VehicleType, Long> perHour) { this.perHour = new EnumMap<>(perHour); }
    public long fee(VehicleType type, Duration stayed) {
        long hours = Math.max(1, (stayed.toMinutes() + 59) / 60);  // started hours
        return hours * perHour.get(type);
    }
}

record FlatPricing(long amount) implements PricingStrategy {
    public long fee(VehicleType type, Duration stayed) { return amount; }
}

final class ParkingLot {
    private final Map<Integer, ParkingFloor> floors = new LinkedHashMap<>();
    private final Clock clock;
    private volatile PricingStrategy pricing;
    private final Map<String, Ticket> active = new ConcurrentHashMap<>();
    private final Set<String> platesInside = ConcurrentHashMap.newKeySet();
    private final AtomicInteger nextTicket = new AtomicInteger(1);

    ParkingLot(List<ParkingFloor> floors, PricingStrategy pricing, Clock clock) {
        floors.forEach(f -> this.floors.put(f.number(), f));
        this.pricing = Objects.requireNonNull(pricing);
        this.clock = Objects.requireNonNull(clock);
    }

    void changePricing(PricingStrategy p) { pricing = Objects.requireNonNull(p); }

    Optional<Ticket> park(Vehicle v) {
        if (!platesInside.add(v.plate()))                  // atomic check-and-add
            throw new IllegalStateException(v.plate() + " is already inside");
        for (ParkingFloor floor : floors.values()) {
            Optional<ParkingSpot> spot = floor.claim(v);
            if (spot.isPresent()) {
                var t = new Ticket("T-%03d".formatted(nextTicket.getAndIncrement()), v,
                        floor.number(), spot.get().id(), clock.instant(), pricing);
                active.put(t.id(), t);
                return Optional.of(t);
            }
        }
        platesInside.remove(v.plate());                    // full: undo
        return Optional.empty();
    }

    Receipt leave(String ticketId) {
        Ticket t = active.remove(ticketId);                // first: one use only
        if (t == null) throw new IllegalArgumentException("unknown or used ticket " + ticketId);
        floors.get(t.floor()).release(t.spotId());
        platesInside.remove(t.vehicle().plate());
        Duration stayed = Duration.between(t.entry(), clock.instant());
        return new Receipt(t.id(), t.vehicle().plate(), t.spotId(), stayed,
                t.pricing().fee(t.vehicle().type(), stayed));
    }

    List<String> availability() {
        return floors.values().stream().map(ParkingFloor::freeSummary).toList();
    }
}

final class MutableClock extends Clock {                   // a clock the demo moves
    private Instant now;
    MutableClock(Instant start) { now = start; }
    void advance(Duration d) { now = now.plus(d); }
    @Override public Instant instant() { return now; }
    @Override public ZoneId getZone() { return ZoneOffset.UTC; }
    @Override public Clock withZone(ZoneId zone) { throw new UnsupportedOperationException(); }
}

public class ParkingLotDemo {
    static final Map<String, String> ticketOf = new HashMap<>();   // plate -> id

    static void park(ParkingLot lot, Vehicle v) {
        String result = lot.park(v).map(t -> {
            ticketOf.put(v.plate(), t.id());
            return t.id() + " at " + t.spotId();
        }).orElse("lot full for " + v.type());
        System.out.println("park  " + v.plate() + ": " + result);
    }

    public static void main(String[] args) {
        var clock = new MutableClock(Instant.parse("2026-10-02T09:00:00Z"));
        var lot = new ParkingLot(List.of(
                new ParkingFloor(1, Map.of(SpotSize.SMALL, 1, SpotSize.MEDIUM, 1)),
                new ParkingFloor(2, Map.of(SpotSize.MEDIUM, 1, SpotSize.LARGE, 1))),
            new HourlyPricing(Map.of(VehicleType.MOTORBIKE, 10L,
                    VehicleType.CAR, 40L, VehicleType.TRUCK, 100L)),
            clock);

        var bike = new Vehicle("BIKE-1", VehicleType.MOTORBIKE);
        park(lot, bike);
        park(lot, new Vehicle("CAR-A", VehicleType.CAR));
        park(lot, new Vehicle("CAR-B", VehicleType.CAR));
        park(lot, new Vehicle("TRUCK-1", VehicleType.TRUCK));
        park(lot, new Vehicle("CAR-C", VehicleType.CAR));        // no room

        clock.advance(Duration.ofMinutes(130));
        System.out.println("leave " + lot.leave(ticketOf.get("CAR-A")));
        clock.advance(Duration.ofMinutes(40));
        System.out.println("leave " + lot.leave(ticketOf.get("TRUCK-1")));

        lot.changePricing(new FlatPricing(250));                 // event starts
        park(lot, new Vehicle("CAR-C", VehicleType.CAR));
        park(lot, new Vehicle("CAR-D", VehicleType.CAR));        // takes LARGE
        try { lot.park(bike); }
        catch (IllegalStateException e) { System.out.println("park  rejected: " + e.getMessage()); }
        System.out.println(lot.availability());

        clock.advance(Duration.ofHours(3));
        System.out.println("leave " + lot.leave(ticketOf.get("CAR-C")));   // flat
        System.out.println("leave " + lot.leave(ticketOf.get("CAR-B")));   // hourly
        try { lot.leave(ticketOf.get("CAR-B")); }
        catch (IllegalArgumentException e) { System.out.println("leave rejected: " + e.getMessage()); }
        System.out.println(lot.availability());
    }
}
